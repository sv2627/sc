1 · Практика 1 — Row і Column: шапка і рейтинг ⭐–⭐⭐⭐ 
// Урок 18 · практика 1 · Row і Column: шапка картки і рейтинг.
// Усе ⭐–⭐⭐⭐ разом. Рамки й кольорів картки ще немає — це практика 2.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      home: Scaffold(
        appBar: AppBar(title: const Text('Game store'), centerTitle: true),
        body: Center(
          // Column — діти згори вниз. Головна вісь — вертикаль.
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              // ⭐ Row — діти зліва направо. Головна вісь — горизонталь.
              const Row(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  Icon(Icons.sports_esports, size: 48, color: Colors.indigo),
                  SizedBox(width: 12), // проміжок 12 між іконкою і текстом
                  // Column у Row: назва над жанром, обидва від лівого краю
                  Column(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      Text('Stardew Valley'),
                      Text('Farming sim'),
                    ],
                  ),
                ],
              ),
              const SizedBox(height: 16), // проміжок 16 між рядками
              // ⭐⭐ Рейтинг: чотири повні зірки, половинка і число
              const Row(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  Icon(Icons.star, color: Colors.amber),
                  Icon(Icons.star, color: Colors.amber),
                  Icon(Icons.star, color: Colors.amber),
                  Icon(Icons.star, color: Colors.amber),
                  Icon(Icons.star_half, color: Colors.amber),
                  SizedBox(width: 8),
                  Text('4.8'),
                ],
              ),
              const SizedBox(height: 16),
              // ⭐⭐⭐ Ціна ліворуч, кошик праворуч. SizedBox дає рядку ширину 300,
              // spaceBetween розкидає дітей по краях цих 300.
              const SizedBox(
                width: 300,
                child: Row(
                  mainAxisAlignment: MainAxisAlignment.spaceBetween,
                  children: [
                    Text('14.99 USD'),
                    Icon(Icons.shopping_cart),
                  ],
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}
2 · Практика 2 — Container і TextStyle: шапка в рамці ⭐–⭐⭐⭐ 
// Урок 18 · практика 2 · Container і TextStyle: шапка гри в рамці.
// Усе ⭐–⭐⭐⭐ разом. Рейтинг і ціну додаємо в практиці 3.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      home: Scaffold(
        backgroundColor: Colors.grey.shade200, // сірий фон — біла картка на ньому видна
        appBar: AppBar(title: const Text('Game store'), centerTitle: true),
        body: Center(
          // ⭐ Container — коробка: розмір, відступи, фон, рамка
          child: Container(
            width: 320,
            padding: const EdgeInsets.all(16), // відступ ВСЕРЕДИНІ: від рамки до тексту
            margin: const EdgeInsets.all(16), // відступ ЗЗОВНІ: від рамки до сусідів
            // ⭐⭐ decoration — фон, заокруглення, рамка. color — лише тут, усередині
            decoration: BoxDecoration(
              color: Colors.white,
              borderRadius: BorderRadius.circular(16),
              border: Border.all(color: Colors.indigo, width: 2),
            ),
            child: const Row(
              children: [
                Icon(Icons.sports_esports, size: 48, color: Colors.indigo),
                SizedBox(width: 12),
                Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    // ⭐ TextStyle — розмір, товщина, колір тексту
                    Text(
                      'Stardew Valley',
                      style: TextStyle(fontSize: 22, fontWeight: FontWeight.bold),
                    ),
                    // ⭐⭐⭐ курсив, приглушений колір, розрідження літер
                    Text(
                      'Farming sim',
                      style: TextStyle(
                        color: Colors.black54,
                        fontStyle: FontStyle.italic,
                        letterSpacing: 1.5,
                      ),
                    ),
                  ],
                ),
              ],
            ),
          ),
        ),
      ),
    );
  }
}
