// Домашнє · урок 10 · «Лічильник води». Прізвище: ...
// Це стандартний лічильник — той самий, що на dartpad.dev/?sample=counter.
// 1. Заголовок Water (рядок home: ...), підпис над числом Glasses today:
// 2. Кнопка -1 під числом, як на уроці, — але не нижче 0.
//    Підказка: усередині setState — if (_counter > 0) { _counter--; }
// Перевір: тричі + -> 3; потім чотири рази -1 -> 0, не -1.

import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Flutter Demo',
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.blue),
      home: const MyHomePage(title: 'Flutter Demo Home Page'),
    );
  }
}

class MyHomePage extends StatefulWidget {
  final String title;

  const MyHomePage({super.key, required this.title});

  @override
  State<MyHomePage> createState() => _MyHomePageState();
}

class _MyHomePageState extends State<MyHomePage> {
  int _counter = 0;

  void _incrementCounter() {
    setState(() {
      _counter++;
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text(widget.title)),
      body: Center(
        child: Column(
          mainAxisAlignment: .center,
          children: [
            const Text('You have pushed the button this many times:'),
            Text(
              '$_counter',
              style: Theme.of(context).textTheme.headlineMedium,
            ),
          ],
        ),
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: _incrementCounter,
        tooltip: 'Increment',
        child: const Icon(Icons.add),
      ),
    );
  }
}

// Урок 10 · Перша програма на Flutter — лічильник після практик 1 і 2.
// Відкрий dartpad.dev, встав увесь код замість того, що там є, натисни Run.
import 'package:flutter/material.dart';

// Точка входу. runApp показує на екрані головний віджет — MyApp.
void main() => runApp(const MyApp());

// MyApp — рамка застосунку: назва, колір, перший екран.
// StatelessWidget — віджет, який сам ніколи не змінюється.
class MyApp extends StatelessWidget {
  const MyApp({super.key});

  // build — «як мене намалювати». Повертає дерево віджетів.
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'My counter',
      debugShowCheckedModeBanner: false,
      // Практика 1: свій колір. Спробуй Colors.red, Colors.purple, Colors.teal.
      theme: ThemeData(colorSchemeSeed: Colors.green),
      // Практика 1: свій заголовок — він потрапить у AppBar.
      home: const MyHomePage(title: 'My counter'),
    );
  }
}

// MyHomePage — екран, який змінюється (число росте).
// StatefulWidget — віджет зі станом. Сам стан живе в класі нижче.
class MyHomePage extends StatefulWidget {
  final String title;

  const MyHomePage({super.key, required this.title});

  @override
  State<MyHomePage> createState() => _MyHomePageState();
}

// Тут дані екрана і код, який їх змінює.
class _MyHomePageState extends State<MyHomePage> {
  int _counter = 0; // стан: число на екрані

  // Кнопка «+». setState — «дані змінились, перемалюй екран».
  void _incrementCounter() {
    setState(() {
      _counter++; // практика 1: крок +5 — це _counter += 5;
    });
  }

  // Практика 2: кнопка «-1». ⭐⭐⭐ — не нижче нуля.
  void _decrementCounter() {
    setState(() {
      if (_counter > 0) {
        _counter--;
      }
    });
  }

  // Практика 2 ⭐⭐: кнопка «Reset» — знову 0.
  void _resetCounter() {
    setState(() {
      _counter = 0;
    });
  }

  // Екран: Scaffold — каркас сторінки з шапкою, тілом і круглою кнопкою.
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text(widget.title)), // widget.title — з MyHomePage
      body: Center(
        // Column — віджети один під одним. .center — по центру екрана.
        child: Column(
          mainAxisAlignment: .center,
          children: [
            const Text('Taps:'), // практика 1: свій текст
            Text(
              '$_counter', // число стає текстом через $
              style: Theme.of(context).textTheme.headlineMedium,
            ),
            // Практика 2: дві нові кнопки під числом.
            ElevatedButton(
              onPressed: _decrementCounter, // без () — передаємо, а не викликаємо
              child: const Text('-1'),
            ),
            ElevatedButton(
              onPressed: _resetCounter,
              child: const Text('Reset'),
            ),
          ],
        ),
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: _incrementCounter,
        tooltip: 'Increment',
        child: const Icon(Icons.add), // практика 1: Icons.favorite, Icons.star
      ),
    );
  }
}

// Урок 10 · Практика 3 «Табло матчу» — ⭐⭐⭐, повна версія.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.orange),
      home: const ScorePage(),
    );
  }
}

class ScorePage extends StatefulWidget {
  const ScorePage({super.key});

  @override
  State<ScorePage> createState() => _ScorePageState();
}

class _ScorePageState extends State<ScorePage> {
  // ⭐ Два числа замість одного — два поля стану.
  int _home = 0;
  int _away = 0;
  final int _win = 5; // ⭐⭐⭐ гра до 5 очок

  void _addHome() {
    setState(() {
      _home++;
    });
  }

  void _addAway() {
    setState(() {
      _away++;
    });
  }

  // ⭐⭐ Кругла кнопка тепер скидає рахунок.
  void _reset() {
    setState(() {
      _home = 0;
      _away = 0;
    });
  }

  // ⭐⭐⭐ Чи закінчилась гра.
  bool _isOver() => _home >= _win || _away >= _win;

  // ⭐⭐⭐ Рядок під рахунком — звичайна функція з if і return.
  String _status() {
    if (_home >= _win) return 'Home wins!';
    if (_away >= _win) return 'Away wins!';
    if (_home > _away) return 'Home leads';
    if (_away > _home) return 'Away leads';
    return 'Draw';
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Scoreboard')),
      body: Center(
        child: Column(
          mainAxisAlignment: .center,
          children: [
            // ⭐⭐ Рахунок одним рядком, великим шрифтом.
            Text(
              '$_home : $_away',
              style: Theme.of(context).textTheme.displayMedium,
            ),
            Text(_status()),
            // ⭐⭐ Row — віджети в рядок, поруч.
            Row(
              mainAxisAlignment: .center,
              children: [
                // ⭐⭐⭐ onPressed: null — кнопка сіра і не натискається.
                ElevatedButton(
                  onPressed: _isOver() ? null : _addHome,
                  child: const Text('+1 Home'),
                ),
                ElevatedButton(
                  onPressed: _isOver() ? null : _addAway,
                  child: const Text('+1 Away'),
                ),
              ],
            ),
          ],
        ),
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: _reset,
        tooltip: 'Reset',
        child: const Icon(Icons.refresh),
      ),
    );
  }
}
