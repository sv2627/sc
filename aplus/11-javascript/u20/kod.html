script.js · фінальний код уроку 
// Урок 20 · Прізвище Імʼя · рівень: ⭐⭐

// умовний курс уроку: скільки гривень коштує 1 одиниця валюти
const RATES = { EUR: 48, USD: 42, PLN: 11 };
const SIGNS = { EUR: '€', USD: '$', PLN: 'zł' };

const nameInput = document.querySelector('#name');
const greet = document.querySelector('#greet');
const uah = document.querySelector('#uah');
const cur = document.querySelector('#cur');
const result = document.querySelector('#result');
const note = document.querySelector('#note');
const chars = document.querySelector('#chars');

// ── 1. Привітання: .value читаємо ВСЕРЕДИНІ обробника — у момент кліку ──
document.querySelector('#hi').addEventListener('click', () => {
  const name = nameInput.value.trim();     // trim — прибрати пробіли з країв
  if (name === '') {
    greet.textContent = 'Спершу введи імʼя';
  } else {
    greet.textContent = `Привіт, ${name}!`;  // textContent: введене лишається текстом
  }
});

// ── 2–4. Конвертер: одна функція, три способи її викликати ──
function convert() {
  const text = uah.value;                  // завжди РЯДОК, навіть у type="number"
  if (text === '') {                       // Number('') дає 0 — тому перевіряємо рядок
    result.textContent = 'Введи суму в гривнях';
    uah.classList.add('error');
    return;
  }
  const money = Number(text);
  if (Number.isNaN(money) || money < 0) {
    result.textContent = 'Сума — додатне число';
    uah.classList.add('error');
    return;
  }
  uah.classList.remove('error');
  const code = cur.value;                  // 'EUR', 'USD' або 'PLN' — value з <option>
  const res = money / RATES[code];         // квадратні дужки: ключ лежить у змінній
  result.textContent = `${money} грн = ${res.toFixed(2)} ${SIGNS[code]}`;
}

document.querySelector('#calc').addEventListener('click', convert);  // за кнопкою
uah.addEventListener('input', convert);    // живий перерахунок — на кожну цифру
cur.addEventListener('change', convert);   // вибрали іншу валюту

// ── 3. Лічильник символів відгуку ──
note.addEventListener('input', () => {
  const n = note.value.length;
  chars.textContent = `Символів: ${n}`;
  chars.classList.toggle('warn', n > 100); // ⭐⭐ понад 100 — червоним
});
