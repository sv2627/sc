1 · Оголосити і викликати 
// Урок 10 · Практика 1 · оголосити і викликати
// function імʼя(параметри) { тіло } — тіло виконується лише під час виклику.
function hello() {
  console.log('Ласкаво просимо в GameStore!');
}
hello();
hello();   // два виклики — два рядки

function greet(name) {          // name — параметр
  console.log(`Привіт, ${name}!`);
}
greet('Оля');                   // 'Оля' — аргумент
greet('Максим');
greet('Ірина');

function showGame(title, price) {
  console.log(`${title} — ${price} грн`);
}
showGame('Celeste', 449);
showGame('Terraria', 199);
showGame('Portal', 319);
// showGame(449, 'Celeste') → «449 — Celeste грн»: аргументи йдуть у параметри по черзі
// Greet('Оля') → ReferenceError: Greet is not defined — великі й малі літери різні

// ⭐⭐ Цикл усередині функції
function countdown(n) {
  for (let i = n; i >= 1; i--) {
    console.log(i);
  }
  console.log('Старт!');
}
countdown(3);   // 3 2 1 Старт!

// ⭐⭐⭐ Таблиця множення на n
function table(n) {
  for (let i = 1; i <= 10; i++) {
    console.log(`${n} × ${i} = ${n * i}`);
  }
}
table(7);   // 7 × 1 = 7 … 7 × 10 = 70
2 · Функції, що повертають: return 
// Урок 10 · Практика 2 · return
// return віддає результат туди, де функцію викликали, і одразу виходить з неї.
function square(n) {
  return n * n;
}
console.log(square(5));    // 25
console.log(square(12));   // 144
// без return: console.log(square(5)) → undefined

function isEven(n) {
  return n % 2 === 0;      // порівняння саме дає true або false
}
console.log(isEven(4));    // true
console.log(isEven(7));    // false

function withDiscount(price) {
  return price * 0.9;      // мінус 10 %
}
console.log(withDiscount(500));   // 450

// ⭐⭐ Більше з двох — два return у різних гілках
function max2(a, b) {
  if (a > b) {
    return a;
  }
  return b;
}
console.log(max2(3, 8));   // 8

// ⭐⭐⭐ Квиток за віком
function ticket(age) {
  if (age < 12) {
    return 'дитячий';
  } else if (age < 18) {
    return 'шкільний';
  }
  return 'дорослий';
}
console.log(ticket(9), ticket(14), ticket(30));   // дитячий шкільний дорослий
3 · Пʼять власних функцій: масив — аргумент 
// Урок 10 · Практика 3 · пʼять функцій із тестовими викликами
function total(prices) {
  let sum = 0;               // накопичувач — до циклу, всередині функції
  for (let i = 0; i < prices.length; i++) {
    sum += prices[i];
  }
  return sum;
}
console.log(total([699, 199, 449]));   // 1347
console.log(total([319, 250]));        // 569

function countCheap(prices, limit) {
  let n = 0;
  for (let i = 0; i < prices.length; i++) {
    if (prices[i] < limit) {
      n++;
    }
  }
  return n;
}
console.log(countCheap([699, 199, 449, 250], 300));   // 2
console.log(countCheap([699, 199, 449, 250], 500));   // 3

function hasGame(games, title) {
  return games.includes(title);
}
console.log(hasGame(['Terraria', 'Celeste'], 'Celeste'));   // true
console.log(hasGame(['Terraria', 'Celeste'], 'Portal'));    // false

function withDiscount(price) {
  return price * 0.9;
}
console.log(withDiscount(500));   // 450
console.log(withDiscount(100));   // 90

function isEven(n) {
  return n % 2 === 0;
}
console.log(isEven(10));   // true
console.log(isEven(3));    // false

// total() без аргументу → TypeError: Cannot read properties of undefined (reading 'length')

// ⭐⭐ Функція у функції: спершу total, потім знижка на результат
const cart = [699, 199, 449];
console.log(withDiscount(total(cart)));   // 1212.3

// ⭐⭐⭐ Чек: кожна гра з ціною і рядок «Разом»
function receipt(games, prices) {
  for (let i = 0; i < games.length; i++) {
    console.log(`${games[i]} — ${prices[i]} грн`);
  }
  console.log(`Разом: ${total(prices)} грн`);
}
receipt(['Minecraft', 'Terraria', 'Celeste'], [699, 199, 449]);   // … Разом: 1347 грн
