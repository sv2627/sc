// Урок 11 · Повторення і практика: класи й обʼєкти — Практики 1 і 2, фінальна версія.
// Відкрий dartpad.dev, встав увесь код і натисни Run.
// Товар — сутність, тому class. Склад — список обʼєктів List<Product>.

class Product {
  String name; // назва
  int price; // ціна за штуку, грн
  int count; // скільки штук на складі
  int sold = 0; // ⭐⭐⭐ скільки продано — поле зі значенням одразу

  // Конструктор: this. — значення одразу в поле.
  Product(this.name, this.price, this.count);

  // Рядок для людини — замість Instance of 'Product'.
  String info() => '$name: $price grn, $count pcs';

  // Вартість залишку цього товару.
  int total() => price * count;

  // Продати n штук. Більше, ніж є, — не продаємо і повертаємо false.
  bool sell(int n) {
    if (n > count) return false;
    count -= n;
    sold += n;
    return true;
  }

  // Поповнити склад.
  void restock(int n) {
    count += n;
  }

  // ⭐⭐ Знижка у відсотках: ~/ — ціле ділення, ціна лишається int.
  void discount(int percent) {
    price = price * (100 - percent) ~/ 100;
  }

  // ⭐⭐⭐ Виторг за проданим.
  int revenue() => sold * price;
}

// ===== Практика 2: функції над складом =====

// Вартість усього складу: сума total() кожного товару.
int stockValue(List<Product> xs) {
  int s = 0;
  for (Product p in xs) {
    s += p.total();
  }
  return s;
}

// Найдешевший товар — функція повертає цілий обʼєкт.
Product cheapest(List<Product> xs) {
  Product best = xs[0];
  for (Product p in xs) {
    if (p.price < best.price) best = p;
  }
  return best;
}

// ⭐⭐⭐ Продати за назвою. Спершу any — бо firstWhere без збігу падає.
bool sellByName(List<Product> xs, String name, int n) {
  if (!xs.any((p) => p.name == name)) {
    print('No such product: $name');
    return false;
  }
  Product p = xs.firstWhere((p) => p.name == name);
  return p.sell(n);
}

void main() {
  // --- Практика 1: один товар ---
  Product pen = Product('pen', 25, 40);
  print(pen.info());
  print('Total: ${pen.total()}');
  print('Sell 50: ${pen.sell(50)}'); // стільки немає
  print('Sell 5: ${pen.sell(5)}');
  pen.restock(10);
  print(pen.info());
  pen.discount(20); // ⭐⭐
  print(pen.info());
  print('Revenue: ${pen.revenue()}'); // ⭐⭐⭐ 5 штук × 20 грн

  // --- Практика 2: склад ---
  List<Product> stock = [
    Product('pen', 25, 40),
    Product('notebook', 60, 12),
    Product('backpack', 900, 0),
    Product('marker', 45, 8),
  ];
  for (Product p in stock) {
    print(p.info());
  }
  print('Stock value: ${stockValue(stock)}');
  print('Cheapest: ${cheapest(stock).name}');
  // Чого немає: where — відібрати, map — лише назви, join — в один рядок.
  print('Out of stock: ${stock.where((p) => p.count == 0).map((p) => p.name).join(', ')}');
  // Скільки товарів дорожчі за 50 грн.
  print('More than 50 grn: ${stock.where((p) => p.price > 50).length}');
  // ⭐⭐ Від найдорожчого до найдешевшого — готова формула з уроку 8.
  stock.sort((a, b) => b.price - a.price);
  print(stock.map((p) => p.name).join(', '));
  // ⭐⭐⭐ Продаж за назвою.
  print(sellByName(stock, 'marker', 3));
  print(sellByName(stock, 'marker', 10));
  print(sellByName(stock, 'ruler', 1));
}

// Урок 11 · Маленький проєкт «Бібліотека» — фінальна версія ⭐–⭐⭐⭐.
// Відкрий dartpad.dev, встав увесь код і натисни Run.
// Книжка — сутність: class Book. «Яку книжку хто взяв» — словник: Map<String, String>.

class Book {
  String title; // назва
  String author; // автор
  bool available = true; // на полиці чи видана

  Book(this.title, this.author);

  // Рядок для людини: in — на полиці, out — видана.
  String info() {
    String state = available ? 'in' : 'out';
    return '$title, $author: $state';
  }

  // ⭐ Видати. Уже видана — false, нічого не міняємо.
  bool borrow() {
    if (!available) return false;
    available = false;
    return true;
  }

  // ⭐ Повернути на полицю.
  void giveBack() {
    available = true;
  }
}

// ⭐⭐⭐ Скільки книжок зараз у читача: рахуємо значення словника видач.
int booksOf(Map<String, String> loans, String reader) {
  int c = 0;
  for (String r in loans.values) {
    if (r == reader) c++;
  }
  return c;
}

// ⭐⭐ Видати книжку читачеві й записати у словник видач.
// ⭐⭐⭐ Ліміт: не більше двох книжок на руках.
bool lend(List<Book> lib, Map<String, String> loans, String title, String reader) {
  if (!lib.any((b) => b.title == title)) {
    print('No such book: $title');
    return false;
  }
  if (booksOf(loans, reader) >= 2) {
    print('$reader already has 2 books');
    return false;
  }
  Book b = lib.firstWhere((b) => b.title == title);
  if (!b.borrow()) {
    print('$title is taken');
    return false;
  }
  loans[title] = reader;
  return true;
}

// ⭐⭐ Повернути: книжка на полицю, запис зі словника — геть.
void giveBackBook(List<Book> lib, Map<String, String> loans, String title) {
  if (!loans.containsKey(title)) {
    print('$title was not lent');
    return;
  }
  lib.firstWhere((b) => b.title == title).giveBack();
  loans.remove(title);
}

void main() {
  List<Book> library = [
    Book('Kobzar', 'Shevchenko'),
    Book('Lisova pisnia', 'Ukrainka'),
    Book('Zakhar Berkut', 'Franko'),
    Book('Tyhrolovy', 'Bahrianyi'),
    Book('Kaidasheva simia', 'Nechui-Levytskyi'),
  ];

  // ⭐ Видати двічі ту саму книжку: вдруге не вийде.
  print(library[4].borrow());
  print(library[4].borrow());
  for (Book b in library) {
    print(b.info());
  }
  print('Available: ${library.where((b) => b.available).length}');

  // ⭐⭐ Словник видач: назва -> читач.
  Map<String, String> loans = {};
  lend(library, loans, 'Kobzar', 'Olha');
  lend(library, loans, 'Tyhrolovy', 'Ivan');
  lend(library, loans, 'Kobzar', 'Ivan'); // уже видана
  lend(library, loans, 'Eneida', 'Nina'); // такої немає
  lend(library, loans, 'Zakhar Berkut', 'Olha');
  lend(library, loans, 'Lisova pisnia', 'Olha'); // ⭐⭐⭐ третя — ліміт
  loans.forEach((title, reader) => print('$title -> $reader'));
  giveBackBook(library, loans, 'Tyhrolovy');
  giveBackBook(library, loans, 'Eneida');
  lend(library, loans, 'Lisova pisnia', 'Nina');
  print('Available: ${library.where((b) => b.available).length}');

  // ⭐⭐⭐ Хто скільки книжок має — словник-лічильник за значеннями loans.
  Map<String, int> perReader = {};
  for (String r in loans.values) {
    perReader.update(r, (v) => v + 1, ifAbsent: () => 1);
  }
  perReader.forEach((r, n) => print('$r: $n'));
}
