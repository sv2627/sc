element.style

Одна властивість CSS — з JavaScript

script.js
const title = document.querySelector('#title'); title.style.color = '#0253b3'; title.style.background = '#fff3c4'; title.style.padding = '8px 12px'; // те саме, що в Elements: // <h1 id="title" style="color: #0253b3; …">
Після .style. — назва властивості CSS, після = — значення рядком у лапках
JS дописує атрибут style в елемент — видно в F12 → Elements
Атрибут style сильніший за правила зі style.css
🐫
Назви з дефісом

У JS дефіса немає: font-size → fontSize

CSSJavaScript
colorstyle.color
font-sizestyle.fontSize
background-colorstyle.backgroundColor
border-radiusstyle.borderRadius
text-decorationstyle.textDecoration
Дефіс прибрати, наступну літеру — великою. Це camelCase — «верблюжий» запис
style.font-size JavaScript читає як мінус: style.font - size → помилка
Розміри — з одиницями: '24px', а не 24. З числа: `${size}px`
🔬
Тонкощі style

style бачить лише те, що на самому елементі

Console · у style.css: #title { color: #591429; }
const t = document.querySelector('#title'); t.style.color // '' — колір у style.css, не на елементі t.style.color = 'red'; t.style.color // 'red' t.style.color = ''; // прибрали — знову бордовий зі style.css t.style.fontSize = 30; // без 'px' — браузер мовчки ігнорує
Порожній рядок '' — не помилка: просто на елементі цієї властивості немає
Скинути — присвоїти ''. Повертається те, що каже CSS
Неправильне значення — без червоного: нічого не змінилось. Дивись у Elements

classList.add · remove

Додати клас — прибрати клас

script.js
const first = document.querySelector('.game'); first.classList.add('hit'); // <p class="game hit">Minecraft… const terra = document.querySelector('.sale'); terra.classList.remove('sale'); // <p class="game">Terraria… — жовтого тла немає
Назва класу — без крапки: add('hit'), не add('.hit'). Крапка — лише в адресі
Решта класів лишається: game нікуди не дівся
Додати двічі — клас буде один. Прибрати неіснуючий — не помилка

Console · виконуй по рядку

toggle — як вимикач світла: кожен виклик міняє стан. Повертає, чи клас тепер є

Умова + дата

Після 18:00 — нічна тема сама

script.js · у кінець
const hour = new Date().getHours(); if (hour >= 18) { document.body.classList.add('dark'); }
getHours() — година від 0 до 23 з годинника компʼютера
>= 18 — з 18:00 включно. 17:59 — це ще 17
Перевірити вдень: тимчасово const hour = 20; — потім повернути getHours()

toggle(клас, умова)

Ніч — і ввечері, і вночі: один рядок

script.js
const hour = new Date().getHours(); const night = hour >= 18 || hour < 7; // true / false document.body.classList.toggle('dark', night); // те саме через if / else: // if (night) add('dark') else remove('dark')
Друга умова — вночі теж: 0–6 година. || — «хоч одна правдива» (урок 3)
toggle('dark', true) — додати, toggle('dark', false) — прибрати. Перемикання за умовою
Умову зручно тримати у змінній з назвою: night читається, як слово

querySelectorAll + for

Змінити усі ігри — цикл по списку

script.js
const games = document.querySelectorAll('.game'); for (let i = 0; i < games.length; i++) { games[i].classList.add('card'); } // або коротше — for...of, як по масиву for (const g of games) { g.style.padding = '6px 10px'; }
querySelectorAll — список елементів: .length, [0], [i] — як у масиву з уроку 9
games.classList.add(…) без циклу — TypeError: у списку немає classList, він є лише в кожного елемента
🏷️
Умова в циклі

Дешеві ігри — зеленою смугою

script.js · у style.css: .cheap { border-left: 4px solid green; }
const prices = [649, 319, 199]; // у тому ж порядку, що ігри const games = document.querySelectorAll('.game'); for (let i = 0; i < games.length; i++) { if (prices[i] < 300) { games[i].classList.add('cheap'); } }
Той самий індекс i — та сама гра: games[2] — Terraria, prices[2] — 199
Тут смугу отримає лише Terraria

Виправ — не запускай, знайди очима

Нічна тема не вмикається

script.js · зламано навмисно
const hour = new Date().getHours(); if (hour >= 18 && hour < 7) { document.body.classList.add('.dark'); } const games = document.querySelectorAll('.game'); games.style.color = 'gray';

Три біди

Знайди всі три, потім клацни

1. && — умова не буває правдивою ніколи. Треба ||

2. '.dark' — з крапкою: додасться клас «.dark», а правило .dark його не впізнає. Треба 'dark'

3. games.style — у списку немає style: TypeError. Потрібен цикл і games[i].style
