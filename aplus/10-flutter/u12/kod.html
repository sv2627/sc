// Домашнє · урок 12 · «Гучність». Прізвище: ...
// Кнопка +1 уже працює (до 10). Допиши два місця з позначкою ДОПИШИ ТУТ.
// Перевір: п'ять разів -1 -> 0 і Mute; ще раз -1 -> лишається 0;
//          вісім разів +1 -> 8 і Loud; ще три рази +1 -> 10.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.blue),
      home: const VolumePage(),
    );
  }
}

class VolumePage extends StatefulWidget {
  const VolumePage({super.key});

  @override
  State<VolumePage> createState() => _VolumePageState();
}

class _VolumePageState extends State<VolumePage> {
  int _volume = 5; // гучність від 0 до 10

  // Готово: +1, але не більше 10.
  void _up() {
    setState(() {
      if (_volume < 10) {
        _volume++;
      }
    });
  }

  // 1. ДОПИШИ ТУТ: -1, але не нижче 0.
  //    Так само, як _up вище: setState, усередині if (_volume > 0) і _volume--.
  void _down() {
  }

  // 2. ДОПИШИ ТУТ: рядок під числом.
  //    0 -> 'Mute', 8 і більше -> 'Loud', інакше — як зараз.
  //    Підказка: два рядки if (...) return '...'; над return нижче.
  String _status() {
    return 'Volume: $_volume';
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Volume')),
      body: Center(
        child: Column(
          mainAxisAlignment: .center,
          children: [
            Text(
              '$_volume',
              style: Theme.of(context).textTheme.displayMedium,
            ),
            Text(_status()),
            Row(
              mainAxisAlignment: .center,
              children: [
                ElevatedButton(
                  onPressed: _down,
                  child: const Text('-1'),
                ),
                ElevatedButton(
                  onPressed: _up,
                  child: const Text('+1'),
                ),
              ],
            ),
          ],
        ),
      ),
    );
  }
}

Практика 1 · кількість у кошику ⭐–⭐⭐⭐ 
// Урок 12 · Практика 1 «Кількість у кошику» — фінальна версія ⭐–⭐⭐⭐.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.teal),
      home: const CartPage(),
    );
  }
}

class CartPage extends StatefulWidget {
  const CartPage({super.key});

  @override
  State<CartPage> createState() => _CartPageState();
}

class _CartPageState extends State<CartPage> {
  int _qty = 1; // стан: скільки штук у кошику
  final int _price = 45; // ціна однієї штуки, не змінюється

  // «+1». ⭐⭐ — не більше 10: кнопку вимикає onPressed нижче.
  void _add() {
    setState(() {
      _qty++;
    });
  }

  // «-1» — не нижче 1: менше однієї штуки в кошику не буває.
  void _remove() {
    setState(() {
      if (_qty > 1) {
        _qty--;
      }
    });
  }

  // Кругла кнопка — знову одна штука.
  void _reset() {
    setState(() {
      _qty = 1;
    });
  }

  // ⭐⭐⭐ Від 5 штук — по 40 грн замість 45.
  int _total() => _qty >= 5 ? _qty * 40 : _qty * _price;

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Cart')),
      body: Center(
        child: Column(
          mainAxisAlignment: .center,
          children: [
            const Text('Coffee beans, 45 grn'),
            Text(
              'Qty: $_qty',
              style: Theme.of(context).textTheme.headlineMedium,
            ),
            // ${...} — у рядку обчислюється вираз, а не просто змінна.
            Text('Total: ${_total()} grn'),
            Text(_qty >= 5 ? 'Discount!' : ''), // ⭐⭐⭐
            Row(
              mainAxisAlignment: .center,
              children: [
                ElevatedButton(
                  onPressed: _remove,
                  child: const Text('-1'),
                ),
                // ⭐⭐ На 10 кнопка сіра: onPressed: null.
                ElevatedButton(
                  onPressed: _qty < 10 ? _add : null,
                  child: const Text('+1'),
                ),
              ],
            ),
          ],
        ),
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: _reset,
        tooltip: 'Reset',
        child: const Icon(Icons.refresh),
      ),
    );
  }
}
Практика 2 · вхід до зали ⭐–⭐⭐⭐ 
// Урок 12 · Практика 2 «Вхід до зали» — фінальна версія ⭐–⭐⭐⭐.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      home: const HallPage(),
    );
  }
}

class HallPage extends StatefulWidget {
  const HallPage({super.key});

  @override
  State<HallPage> createState() => _HallPageState();
}

class _HallPageState extends State<HallPage> {
  int _inside = 0; // скільки людей у залі зараз
  int _today = 0; // ⭐⭐ скільки зайшло за день — вихід його не зменшує
  final int _capacity = 5; // місткість зали

  void _enter() {
    setState(() {
      _inside++;
      _today++;
    });
  }

  void _exit() {
    setState(() {
      if (_inside > 0) {
        _inside--;
      }
    });
  }

  // ⭐⭐⭐ Нова зміна — обидва числа знову 0.
  void _newShift() {
    setState(() {
      _inside = 0;
      _today = 0;
    });
  }

  // ⭐⭐ Рядок стану — звичайна функція з if і return.
  String _status() {
    if (_inside == 0) return 'Empty';
    if (_inside >= _capacity) return 'Full';
    return 'Open';
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      // ⭐⭐⭐ Заголовок теж може показувати стан.
      appBar: AppBar(title: Text('Hall $_inside / $_capacity')),
      body: Center(
        child: Column(
          mainAxisAlignment: .center,
          children: [
            Text(
              'Inside: $_inside',
              style: Theme.of(context).textTheme.headlineMedium,
            ),
            Text('Free: ${_capacity - _inside}'),
            Text(_status()),
            Text('Today: $_today'),
            Row(
              mainAxisAlignment: .center,
              children: [
                // Зала повна — вхід вимкнено.
                ElevatedButton(
                  onPressed: _inside < _capacity ? _enter : null,
                  child: const Text('Enter'),
                ),
                ElevatedButton(
                  onPressed: _exit,
                  child: const Text('Exit'),
                ),
              ],
            ),
          ],
        ),
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: _newShift,
        tooltip: 'New shift',
        child: const Icon(Icons.refresh),
      ),
    );
  }
}
Маленький проєкт · каса кавʼярні ⭐–⭐⭐⭐ 
// Урок 12 · Маленький проєкт «Каса кавʼярні» — фінальна версія ⭐–⭐⭐⭐.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.brown),
      home: const CashPage(),
    );
  }
}

class CashPage extends StatefulWidget {
  const CashPage({super.key});

  @override
  State<CashPage> createState() => _CashPageState();
}

class _CashPageState extends State<CashPage> {
  // ⭐ Поточне замовлення: сума і скільки позицій.
  int _total = 0;
  int _items = 0;
  // ⭐⭐ Що замовили: словник «назва -> скільки штук».
  Map<String, int> _order = {};
  // ⭐⭐⭐ Підсумок дня: скільки оплачених замовлень і виторг.
  int _paid = 0;
  int _revenue = 0;

  // ⭐⭐⭐ Один метод на всі кнопки: назва і ціна приходять параметрами.
  // На рівні ⭐ тут були три методи: _addCoffee, _addTea, _addCroissant.
  void _add(String name, int price) {
    setState(() {
      _total += price;
      _items++;
      _order.update(name, (v) => v + 1, ifAbsent: () => 1); // ⭐⭐
    });
  }

  // ⭐ Кругла кнопка — очистити замовлення.
  void _clear() {
    setState(() {
      _total = 0;
      _items = 0;
      _order.clear();
    });
  }

  // ⭐⭐ Замовлення одним рядком: coffee x2, tea x1
  String _orderText() {
    if (_order.isEmpty) return 'Empty';
    return _order.keys.map((k) => '$k x${_order[k]}').join(', ');
  }

  // ⭐⭐⭐ Від 200 грн — знижка 10 %.
  int _toPay() => _total >= 200 ? _total * 90 ~/ 100 : _total;

  // ⭐⭐⭐ Оплата: замовлення йде в підсумок дня, каса знову порожня.
  void _pay() {
    setState(() {
      _paid++;
      _revenue += _toPay();
      _total = 0;
      _items = 0;
      _order.clear();
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Cafe')),
      body: Center(
        child: Column(
          mainAxisAlignment: .center,
          children: [
            Text(
              '$_total grn',
              style: Theme.of(context).textTheme.displayMedium,
            ),
            Text('Items: $_items'),
            Text(_orderText()),
            Text('To pay: ${_toPay()} grn'),
            Row(
              mainAxisAlignment: .center,
              children: [
                // () => _add(...) — функція без імені, як ifAbsent: () => 1.
                ElevatedButton(
                  onPressed: () => _add('coffee', 45),
                  child: const Text('Coffee 45'),
                ),
                ElevatedButton(
                  onPressed: () => _add('tea', 30),
                  child: const Text('Tea 30'),
                ),
                ElevatedButton(
                  onPressed: () => _add('croissant', 55),
                  child: const Text('Croissant 55'),
                ),
              ],
            ),
            // ⭐⭐⭐ Порожнє замовлення оплатити не можна.
            ElevatedButton(
              onPressed: _items == 0 ? null : _pay,
              child: const Text('Pay'),
            ),
            Text('Orders today: $_paid, revenue: $_revenue grn'),
          ],
        ),
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: _clear,
        tooltip: 'Clear',
        child: const Icon(Icons.refresh),
      ),
    );
  }
}
