// Урок 19 · Прізвище Імʼя · рівень: ⭐⭐⭐

const games = [
  { name: 'Minecraft', price: 699, genre: 'пісочниця' },
  { name: 'Stardew Valley', price: 319, genre: 'ферма' },
  { name: 'Terraria', price: 199, genre: 'пригоди' },
  { name: 'Celeste', price: 449, genre: 'платформер' },
  { name: 'Portal 2', price: 279, genre: 'головоломка' },
];
const catalog = document.querySelector('#catalog');
const info = document.querySelector('#info');
const msg = document.querySelector('#msg');

// render: тепер кожна картка ще й слухає клік
function render(list) {
  catalog.innerHTML = '';                 // перемалювати, а не дописати
  for (const g of list) {
    const card = document.createElement('div');
    card.classList.add('card');
    if (g.price < 300) card.classList.add('sale');
    const h = document.createElement('h3');
    h.textContent = g.name;
    const price = document.createElement('p');
    price.classList.add('price');
    price.textContent = `${g.price} грн`;
    card.append(h, price);

    // Практика 3: свій обробник у кожної картки — у ньому її card і її g
    card.addEventListener('click', () => {
      card.classList.toggle('picked');
      msg.textContent = `Обрано: ${g.name}`;
    });
    // ⭐⭐⭐ подвійний клік прибирає картку
    card.addEventListener('dblclick', () => card.remove());

    catalog.append(card);
  }
  info.textContent = `Ігор: ${list.length}`;
}

render(games);

// ── 2. Лічильник: змінна ПОЗА обробниками, інакше щоразу з нуля ──
const out = document.querySelector('#count');
let count = 0;

function show() {                          // ⭐⭐⭐ одна функція оновлює текст
  out.textContent = `Кліків: ${count}`;
  out.classList.toggle('gold', count >= 10);
}
document.querySelector('#plus').addEventListener('click', () => {
  count++;
  show();
});
document.querySelector('#minus').addEventListener('click', () => {
  if (count > 0) count--;                  // ⭐⭐ нижче нуля не можна
  show();
});
document.querySelector('#reset').addEventListener('click', () => {
  count = 0;
  show();
});

// ── 3. Тема: toggle вмикає/вимикає, contains питає, яка зараз ──
const theme = document.querySelector('#theme');
theme.addEventListener('click', () => {
  document.body.classList.toggle('dark');
  if (document.body.classList.contains('dark')) {
    theme.textContent = '☀️ Світла';
  } else {
    theme.textContent = '🌙 Темна';
  }
});

// ⭐⭐ заголовок більшає, поки над ним курсор
const title = document.querySelector('#title');
title.addEventListener('mouseenter', () => title.classList.add('big'));
title.addEventListener('mouseleave', () => title.classList.remove('big'));

// ── 4. Фільтр: render з потрібним масивом; стрілкова, бо треба передати аргумент ──
const cheapGames = [];
for (const g of games) {
  if (g.price < 300) cheapGames.push(g);
}
const all = document.querySelector('#all');
const cheap = document.querySelector('#cheap');

all.addEventListener('click', () => {
  render(games);
  all.classList.add('on');                 // ⭐⭐ підсвітити натиснуту
  cheap.classList.remove('on');
});
cheap.addEventListener('click', () => {
  render(cheapGames);
  cheap.classList.add('on');
  all.classList.remove('on');
});
