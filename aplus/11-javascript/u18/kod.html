// Урок 18 · Прізвище Імʼя · рівень: ⭐⭐⭐

const games = [
  { name: 'Minecraft', price: 699, genre: 'пісочниця' },
  { name: 'Stardew Valley', price: 319, genre: 'ферма' },
  { name: 'Terraria', price: 199, genre: 'пригоди' },
  { name: 'Celeste', price: 449, genre: 'платформер' },
  { name: 'Portal 2', price: 279, genre: 'головоломка' },
];
const catalog = document.querySelector('#catalog');   // з # — інакше null
const info = document.querySelector('#info');

// ── 1. Обʼєкти: крапка + назва властивості ──
console.table(games);
console.log(games[2].name);   // Terraria

// ── 2. Спосіб 1: innerHTML += з шаблонним рядком (лише для СВОЇХ даних) ──
// for (const g of games) {
//   catalog.innerHTML += `<div class="card"><h3>${g.name}</h3><p class="price">${g.price} грн</p></div>`;
// }

// ── 3–4. Спосіб 2: createElement + append, загорнуто у функцію render ──
function render(list) {
  catalog.innerHTML = '';     // стерти старі картки, щоб не дублювались
  let sum = 0;
  let cheap = 0;
  for (const g of list) {
    const card = document.createElement('div');   // 1) створити в памʼяті
    card.classList.add('card');
    if (g.price < 300) {
      card.classList.add('sale');
      cheap++;
    }
    const h = document.createElement('h3');
    h.textContent = g.name;                         // 2) заповнити — textContent
    const genre = document.createElement('p');
    genre.textContent = g.genre;
    const price = document.createElement('p');
    price.classList.add('price');
    price.textContent = `${g.price} грн`;
    card.append(h, genre, price);
    catalog.append(card);                           // 3) вставити на сторінку
    sum += g.price;
  }
  info.textContent = `Ігор: ${list.length} · разом ${sum} грн · дешевих: ${cheap}`;
}

render(games);   // Ігор: 5 · разом 1945 грн · дешевих: 2

// Вітрина дешевих: новий масив через push — і та сама функція
const cheapGames = [];
for (const g of games) {
  if (g.price < 300) cheapGames.push(g);
}

// ⭐⭐⭐ Ігри одного жанру — функція повертає новий масив
function byGenre(genre) {
  const result = [];
  for (const g of games) {
    if (g.genre === genre) result.push(g);
  }
  return result;
}
