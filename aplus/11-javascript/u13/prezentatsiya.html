Приклад
function shout(text) { return text.trim() .toUpperCase() + '!'; } console.log(shout(' ура ')); // УРА!
Тепер самостійно — у консоль
  • Набери shout і перевір на ' привіт '
  • spell(word) — після кожної літери крапка: 'кіт' → 'к.і.т.'
  • countVowels(text) — масив ['а', 'е', 'и', 'і', 'о', 'у'] і includes(ch). 'молоко' → 3
  • reverse(word) — задом наперед: 'літо' → 'отіл'
  • ⭐⭐ longest(words) — найдовше слово масиву: ['сон', 'вітер', 'дощ'] → 'вітер'
  • ⭐⭐⭐ censor(text, letter) — літеру замінити на *: ('банан', 'а') → 'б*н*н'

function hint(guess, secret) { if (guess > secret) return 'Менше'; if (guess < secret) return 'Більше'; return 'Вгадав!'; } const secret = Math.floor(Math.random() * 100) + 1; let tries = 0; let guess; do { const raw = prompt('Число від 1 до 100'); if (raw === null) break; guess = Number(raw); tries++; alert(hint(guess, secret)); } while (guess !== secret);
hint лише повертає слово. Показує його alert
do...while — бо спитати треба хоч раз
Скасувати → null → break. Без цього вікно не відпустить
Перевіряти зручніше, коли знаєш число: console.log('secret =', secret)

const fruits = ['яблуко', 'груша', 'слива']; function randomItem(arr) { const i = Math.floor(Math.random() * arr.length) + 1; arr[i]; } function shout(text) { return text.toUppercase() + '!'; } console.log(shout(randomItem(fruits)));
КонсольTypeError: Cannot read properties of undefined (reading 'toUppercase')

Три біди — лагодь по одній

Знайди всі три, потім клацни

1. У randomItem немає return — функція дає undefined. Треба return arr[i];

2. toUppercase — такого немає: TypeError … is not a function. Треба toUpperCase

3. + 1 — індекс 1…3: «яблуко» не випадає ніколи, а на 3 знову undefined. Прибрати
