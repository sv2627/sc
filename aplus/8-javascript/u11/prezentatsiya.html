Виправ · console.log і breakpoint

Має бути 29, а в консолі NaN

script.js · зламано навмисно
function sum(marks) { let s = 0; for (let i = 0; i <= marks.length; i++) { s += marks[i]; } return s; } console.log(sum([10, 7, 12]));
КонсольNaN

Як знайти? Клацни

console.log: у цикл перед s += … додай console.log(i, marks[i]); — останній рядок 3 undefined

breakpoint: F12 → Sources → script.js → клік по номеру рядка 4 → оновити сторінку → дивимось i і s на кожному кроці

Що не так? Клацни

i <= marks.length заходить на marks[3] — його немає, це undefined. 29 + undefined = NaN. Межа циклу — <

Уся статистика на одному екрані

sum — сума

let s = 0;
for (…) s += arr[i];
return s;

average — середнє

return sum(arr) / arr.length;
// вивід:
avg.toFixed(1)   // '8.8'

max · min

let m = arr[0];
if (arr[i] > m) m = arr[i];
// min — знак <

скільки — лічильник

let n = 0;
if (arr[i] >= 10) n++;
return n;

новий масив

const res = [];
if (a[i] >= 7) res.push(a[i]);
return res;

sort — порядок

a.sort((x, y) => x - y); // ↑
a.sort((x, y) => y - x); // ↓
