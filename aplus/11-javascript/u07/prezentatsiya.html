let c1 = prompt('1 — ліс, 2 — гори'); if (c1 !== null) { c1 = c1.trim().toLowerCase(); } if (c1 === null) { alert('Гру перервано.'); } else if (c1 === '1' || c1 === 'ліс') { alert('Ліс густий і тихий.'); } else if (c1 === '2' || c1 === 'гори') { alert('У горах холодний вітер.'); } else { alert('Ти вагався до темряви. Цифра або слово

|| — і обидва порівняння повністю

пастка — будь-що веде в ліс
if (c1 === '1' || 'ліс') { alert('Ліс.'); }

Праворуч від || стоїть просто рядок 'ліс'. Непорожній рядок — завжди «так». Умова істинна на будь-який ввід

правильно — два порівняння
if (c1 === '1' || c1 === 'ліс') { alert('Ліс.'); }

Кожна половина — окреме питання «c1 дорівнює …?»

let c1, а не const: після prompt ми перезаписуємо його очищеним
Слово в підказці пиши малими: після toLowerCase великих літер уже немає

alert, prompt і confirm

ВикликЩо бачить людинаЩо повертає
alert('…')текст і OKнічого не повертає
prompt('…')текст, поле, OK, Скасуватирядок або null
confirm('…')текст, OK, Скасувати — поля немаєtrue або false

confirm як розвилка

Так / ні — без цифр і без одруківок

script.js
let gold = 0; const agree = confirm('Тут скриня. Відчинити?'); if (agree) { gold += 5; alert(`Монети! Золото: ${gold}`); } else { alert('Ти пройшов повз.'); }
if (agree) — без === true: там і так true чи false
Скасувати тут — не вихід із гри, а відповідь «ні»
Коротше: if (confirm('Відчинити?')) — без змінної
Очищати нічого не треба: людина нічого не вводить

Три біди — і лише одна червона

script.js · зламано навмисно
let score = 0; const agree = confirm('Відчинити двері?'); if (agree === 'true') { score += 10; } let c1 = prompt('1 — вліво, 2 — вправо'); c1 = c1.trim(); if (c1 === '1') { alert('Ліворуч — сад.'); } else { alert('Праворуч — брама.'); } if (confirm('Ще раз?')) { location.reload; }

Знайди всі три

спершу скажи, яка з них червона, потім клацни

1. agree === 'true' — рядок. confirm дає true без лапок, очки не додаються ніколи. Треба if (agree)

2. Скасувати → null.trim() → TypeError. Це і є червона. Спершу if (c1 !== null)

Діагностика

Шість бід сьогоднішнього уроку

Гра замовкла після відповіді — висяча гілка, бракує else
Cannot read properties of null (reading 'trim') — trim до перевірки на null
Будь-яке слово веде в одну гілку — c1 === '1' || 'ліс'
Слово не впізнається — у порівнянні 'Ліс' з великої, а ввід уже малими
Гілка confirm не спрацьовує — порівняння з 'true' або з null
Очки завжди 0 — score + 10; без = нічого не змінює
🔦
Гілка не та? console.log('[' + c1 + ']', typeof c1) одразу після очищення покаже і пробіли, і регістр, і null.
