Оголосити · викликати

Оголошення нічого не запускає

script.js
function hello() { console.log('Привіт!'); } hello(); hello();
КонсольПривіт! Привіт!
Перші три рядки — лише рецепт. Консоль мовчить, доки немає виклику
Виклик — імʼя і дужки (). Скільки викликів — стільки запусків
Без дужок console.log(hello) друкує текст функції, а не запускає її
Хто писав на Python: def → function, відступ → { }
📥
Параметри й аргументи

Функція отримує дані ззовні

script.js
function greet(name, age) { console.log(`Привіт, ${name}! Тобі ${age}.`); } greet('Оля', 16); greet('Максим', 17); greet(16, 'Оля');
КонсольПривіт, Оля! Тобі 16. Привіт, Максим! Тобі 17. Привіт, 16! Тобі Оля.
Параметр — імʼя в дужках оголошення: name, age
Аргумент — значення в дужках виклику: 'Оля', 16
Аргументи стають параметрами за порядком, не за змістом — дивись третій рядок

return

Функція віддає результат

script.js
function square(n) { return n * n; } const s = square(5); console.log(s); // 25 console.log(square(3) + 1); // 10 console.log(square(square(2))); // 16
Виклик square(5) замінюється на те, що стоїть після return
Результат можна покласти в змінну, додати до нього, передати в іншу функцію
Функція, що повертає, — як калькулятор: дала число, а що з ним робити — вирішуєш ти

Головна пастка

return — не те саме, що console.log

надрукувала — і все
function square(n) { console.log(n * n); } const s = square(5); // друкує 25 console.log(s + 1); // NaN
повернула — можна рахувати далі
function square(n) { return n * n; } const s = square(5); // нічого не друкує console.log(s + 1); // 26
Без return функція повертає undefined. А undefined + 1 → NaN
console.log — для людини, return — для програми

return зупиняє

Після return функція закінчується

script.js
function divide(a, b) { if (b === 0) { return 'на нуль не можна'; } return a / b; } console.log(divide(10, 2)); // 5 console.log(divide(1, 0)); // на нуль не можна
Спрацював return — решта рядків функції не виконується
Рання перевірка вгорі замість великого else: погане відсіяли — далі звичайний шлях
Як break у циклі (урок 8), тільки виходить із усієї функції

Так чи ні

Функція-перевірка повертає true або false

script.js
function isEven(n) { return n % 2 === 0; } function isAdult(age) { return age >= 18; } if (isEven(42)) { console.log('парне'); } console.log(isAdult(16)); // false
Назва is… — питання, відповідь на яке «так» або «ні»
return n % 2 === 0 — без if: порівняння вже дає true чи false
Такий виклик стоїть прямо в умові: if (isEven(x))

Кілька return

Оцінка словами — без жодного else

script.js
function grade(score) { if (score >= 10) return 'відмінно'; if (score >= 7) return 'добре'; if (score >= 4) return 'задовільно'; return 'спробуй ще'; } console.log(grade(11)); // відмінно console.log(grade(5)); // задовільно
else не потрібен: return уже вийшов із функції

Область видимості

Що в функції — лишається в функції

локальна — назовні не видно
function makeTotal() { let total = 100; return total * 2; } console.log(makeTotal()); // 200 console.log(total); // ReferenceError: // total is not defined
зовнішня — всередині видно
const RATE = 41.5; function toUah(usd) { return usd * RATE; } console.log(toUah(2)); // 83
Змінна, оголошена в функції, живе лише під час виклику
Параметри — теж локальні
Однакові імена в різних функціях одна одній не заважають

Функціональний вираз

Функцію можна покласти в змінну

script.js
const square = function (n) { return n * n; }; console.log(square(4)); // 16
Функція — значення, як число чи рядок. Її кладуть у змінну
Імені після function немає — імʼя дає змінна
Наприкінці — ;: це звичайне присвоєння const x = …;

Стрілкова функція

Коротший запис: =>

script.js
const square = (n) => n * n; const add = (a, b) => a + b; const hello = () => console.log('Привіт!'); const grade = (score) => { if (score >= 10) return 'відмінно'; return 'спробуй ще'; }; console.log(square(6), add(2, 3)); // 36 5
Одна дія — без { } і без return: значення повертається саме
