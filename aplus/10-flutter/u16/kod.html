1 · Стартовий код · рівень ⭐ уже працює 
// Міні-проєкт II · урок 16 · стартовий код (рівень ⭐ уже працює на книжках).
// Прізвище: ...   Тема: ...   Рівень: ⭐
// Зроби своє: заміни три місця з позначкою ЗАМІНИ — клас, список, тексти на екрані.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

// 1. ЗАМІНИ: свій клас. Сутність своєї теми — 3 поля і конструктор.
//    Книга: назва, автор, сторінки. Твоя тема — свої поля й типи.
class Book {
  final String title;
  final String author;
  final int pages;

  const Book(this.title, this.author, this.pages);
}

// 2. ЗАМІНИ: свій список — 5 або більше обʼєктів твого класу.
//    Порядок важливий, ходимо за номером -> List.
const List<Book> books = [
  Book('The Hobbit', 'J. R. R. Tolkien', 310),
  Book('Animal Farm', 'George Orwell', 112),
  Book('Dune', 'Frank Herbert', 412),
  Book('The Little Prince', 'Antoine de Saint-Exupery', 96),
  Book('Fahrenheit 451', 'Ray Bradbury', 158),
];

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo), // свій колір
      home: const CatalogPage(),
    );
  }
}

class CatalogPage extends StatefulWidget {
  const CatalogPage({super.key});

  @override
  State<CatalogPage> createState() => _CatalogPageState();
}

class _CatalogPageState extends State<CatalogPage> {
  int _index = 0; // номер обʼєкта у списку, з нуля

  // Наступний — не далі останнього.
  void _next() {
    setState(() {
      if (_index < books.length - 1) {
        _index++;
      }
    });
  }

  // Попередній — не раніше першого.
  void _prev() {
    setState(() {
      if (_index > 0) {
        _index--;
      }
    });
  }

  @override
  Widget build(BuildContext context) {
    Book b = books[_index]; // поточний обʼєкт
    return Scaffold(
      // 3. ЗАМІНИ: 'Book' -> назва твоєї сутності. Номер людині — з одиниці.
      appBar: AppBar(title: Text('Book ${_index + 1} / ${books.length}')),
      body: Center(
        child: Column(
          mainAxisAlignment: .center,
          children: [
            // 3. ЗАМІНИ: поля свого класу замість title, author, pages.
            Text(b.title, style: Theme.of(context).textTheme.headlineMedium),
            Text(b.author),
            Text('${b.pages} pages'),
            Row(
              mainAxisAlignment: .center,
              children: [
                // На краях кнопка сіра: null — «натискати нічого».
                ElevatedButton(
                  onPressed: _index > 0 ? _prev : null,
                  child: const Icon(Icons.skip_previous),
                ),
                ElevatedButton(
                  onPressed: _index < books.length - 1 ? _next : null,
                  child: const Icon(Icons.skip_next),
                ),
              ],
            ),
          ],
        ),
      ),
    );
  }
}

2 · Фінальний код прикладу «Книги» ⭐–⭐⭐⭐ 
// Міні-проєкт II · урок 16 · фінальна версія прикладу «Книги», рівні ⭐–⭐⭐⭐.
// Тема своя у кожного — тут лише зразок, як може виглядати кожен рівень.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

// ⭐ Книга — сутність: поля + поведінка -> class.
class Book {
  final String title;
  final String author;
  final int pages;

  const Book(this.title, this.author, this.pages);

  // ⭐⭐ Метод класу: відповідь про саму книгу.
  bool isLong() => pages > 300;

  // ⭐⭐ Метод класу: рядок для екрана.
  String info() => '$author · $pages pages';
}

// ⭐ Порядок важливий, ходимо за номером -> List.
const List<Book> books = [
  Book('The Hobbit', 'J. R. R. Tolkien', 310),
  Book('Animal Farm', 'George Orwell', 112),
  Book('Dune', 'Frank Herbert', 412),
  Book('The Little Prince', 'Antoine de Saint-Exupery', 96),
  Book('Fahrenheit 451', 'Ray Bradbury', 158),
];

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      home: const CatalogPage(),
    );
  }
}

class CatalogPage extends StatefulWidget {
  const CatalogPage({super.key});

  @override
  State<CatalogPage> createState() => _CatalogPageState();
}

class _CatalogPageState extends State<CatalogPage> {
  int _index = 0; // номер у списку, який зараз на екрані, з нуля

  // ⭐⭐ Лайки: словник «назва -> скільки лайків» -> Map.
  final Map<String, int> _likes = {};

  // ⭐⭐⭐ Перемикач: показувати всі книги чи тільки довгі.
  bool _onlyLong = false;

  // ⭐⭐⭐ Список, який зараз на екрані: усі або відфільтровані через where.
  List<Book> _shown() {
    if (_onlyLong) {
      return books.where((b) => b.isLong()).toList();
    }
    return books;
  }

  // ⭐ Наступна — не далі останньої.
  void _next() {
    setState(() {
      if (_index < _shown().length - 1) {
        _index++;
      }
    });
  }

  // ⭐ Попередня — не раніше першої.
  void _prev() {
    setState(() {
      if (_index > 0) {
        _index--;
      }
    });
  }

  // ⭐⭐ Лайк поточній книзі: немає ключа — стане 1, є — +1.
  void _like() {
    setState(() {
      String key = _shown()[_index].title;
      _likes.update(key, (v) => v + 1, ifAbsent: () => 1);
    });
  }

  // ⭐⭐⭐ Перемкнути фільтр. Номер — знову 0: у новому списку інша довжина.
  void _toggleFilter() {
    setState(() {
      _onlyLong = !_onlyLong;
      _index = 0;
    });
  }

  // ⭐⭐ Підсумок циклом: скільки сторінок у списку на екрані.
  int _totalPages() {
    int sum = 0;
    for (Book b in _shown()) {
      sum += b.pages;
    }
    return sum;
  }

  // ⭐⭐⭐ Найпопулярніша: прохід по Map, у кого найбільше лайків.
  String _top() {
    String best = '-';
    int max = 0;
    _likes.forEach((title, n) {
      if (n > max) {
        max = n;
        best = title;
      }
    });
    return best;
  }

  @override
  Widget build(BuildContext context) {
    List<Book> list = _shown();
    Book b = list[_index]; // поточна книга
    return Scaffold(
      // ⭐ Номер людині — з одиниці.
      appBar: AppBar(title: Text('Book ${_index + 1} / ${list.length}')),
      body: Center(
        child: Column(
          mainAxisAlignment: .center,
          children: [
            Text(b.title, style: Theme.of(context).textTheme.headlineMedium),
            Text(b.info()), // ⭐⭐ метод класу
            Text(b.isLong() ? 'Long read' : 'Short read'), // ⭐⭐ метод класу
            Text('Likes: ${_likes[b.title] ?? 0}'), // ⭐⭐ немає в Map -> 0
            Row(
              mainAxisAlignment: .center,
              children: [
                ElevatedButton(
                  onPressed: _index > 0 ? _prev : null,
                  child: const Icon(Icons.skip_previous),
                ),
                ElevatedButton(
                  onPressed: _like,
                  child: const Icon(Icons.favorite),
                ),
                ElevatedButton(
                  onPressed: _index < list.length - 1 ? _next : null,
                  child: const Icon(Icons.skip_next),
                ),
              ],
            ),
            Text('Total: ${_totalPages()} pages'), // ⭐⭐ підсумок
            Text('Top: ${_top()}'), // ⭐⭐⭐ найбільше лайків
          ],
        ),
      ),
      // ⭐⭐⭐ Кругла кнопка — фільтр «тільки довгі» / «усі».
      floatingActionButton: FloatingActionButton(
        onPressed: _toggleFilter,
        tooltip: 'Long only',
        child: Icon(_onlyLong ? Icons.filter_alt_off : Icons.filter_alt),
      ),
    );
  }
}

Позначки ⭐, ⭐⭐, ⭐⭐⭐ у коментарях — до якого рівня належить рядок. Перевір: ♥ ♥ ▶ ▶ ♥ → Book 3 / 5, Likes: 1, Top: The Hobbit; фільтр → Book 1 / 2, Total: 722 pages.
