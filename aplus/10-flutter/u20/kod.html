// У кожної картки свій стан: лайкнув одну гру — інші не змінились.

// Клас 1 — сам віджет. Тут лише те, що приходить ззовні (game), і createState.
class GameCard extends StatefulWidget {
  const GameCard({super.key, required this.game});

  final Game game;

  @override
  State<GameCard> createState() => _GameCardState();
}

// Клас 2 — стан. Тут поля, які змінюються, методи і build.
class _GameCardState extends State<GameCard> {
  bool _liked = false; // приклад: сердечко
  int _inCart = 0; // самостійно: кількість у кошику

  void _toggleLike() {
    setState(() {
      _liked = !_liked; // false -> true -> false ...
    });
  }

  void _add() {
    setState(() {
      _inCart++;
    });
  }

  void _remove() {
    setState(() {
      _inCart--;
    });
  }

  @override
  Widget build(BuildContext context) {
    // ⭐⭐⭐ сума за цю гру; гра — з віджета, тому widget.game
    double total = _inCart * widget.game.price;
    return Container(
      width: 340,
      margin: const EdgeInsets.only(bottom: 12),
      padding: const EdgeInsets.all(16),
      decoration: BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.circular(16),
        // ⭐⭐⭐ є в кошику — зелена рамка
        border: Border.all(color: _inCart > 0 ? Colors.green : Colors.indigo, width: 2),
      ),
      child: Row(
        children: [
          Expanded(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text(
                  widget.game.title, // поле віджета — через widget.
                  style: const TextStyle(fontSize: 20, fontWeight: FontWeight.bold),
                ),
                Text('${widget.game.price} USD'),
                Text('In cart: $_inCart'),
                if (_inCart > 0) Text('Total: ${total.toStringAsFixed(2)} USD'),
              ],
            ),
          ),
          IconButton(
            onPressed: _toggleLike, // без дужок: передаємо метод, а не викликаємо
            icon: Icon(
              _liked ? Icons.favorite : Icons.favorite_border,
              color: Colors.red,
            ),
          ),
          // ⭐⭐ «−» сіра, коли в кошику 0: onPressed: null вимикає кнопку
          IconButton(onPressed: _inCart > 0 ? _remove : null, icon: const Icon(Icons.remove)),
          IconButton(onPressed: _add, icon: const Icon(Icons.add)),
        ],
      ),
    );
  }
}

2 · Переглядач ігор: стан у всього екрана ⭐–⭐⭐⭐ 
// Урок 20 · практика 2 · переглядач ігор: стан — у всього екрана.
// Номер гри _index живе в State екрана; кнопки міняють його через setState.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class Game {
  final String title;
  final String genre;
  final double price;

  const Game(this.title, this.genre, this.price);
}

const List<Game> games = [
  Game('Stardew Valley', 'Farming sim', 14.99),
  Game('Celeste', 'Platformer', 19.99),
  Game('Terraria', 'Sandbox', 9.99),
  Game('Hades II', 'Action', 29.99),
];

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      home: const GameViewer(), // екран — окремий віджет зі станом
    );
  }
}

class GameViewer extends StatefulWidget {
  const GameViewer({super.key});

  @override
  State<GameViewer> createState() => _GameViewerState();
}

class _GameViewerState extends State<GameViewer> {
  int _index = 0; // ⭐ яка гра на екрані
  final List<Game> _cart = []; // ⭐⭐⭐ кошик: сам список final, але вміст міняється

  void _next() {
    setState(() {
      _index++;
    });
  }

  void _prev() {
    setState(() {
      _index--;
    });
  }

  void _buy() {
    setState(() {
      _cart.add(games[_index]); // ⭐⭐⭐
    });
  }

  @override
  Widget build(BuildContext context) {
    Game g = games[_index];
    double sum = 0;
    for (Game c in _cart) {
      sum += c.price;
    }
    return Scaffold(
      backgroundColor: Colors.grey.shade200,
      appBar: AppBar(title: Text('Cart: ${_cart.length}'), centerTitle: true), // ⭐⭐⭐
      body: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            GameCard(game: g),
            Text('${_index + 1} / ${games.length}'), // ⭐⭐ номер гри
            Row(
              mainAxisAlignment: MainAxisAlignment.center,
              children: [
                // ⭐⭐ на першій грі «Prev» сіра, на останній — «Next»
                ElevatedButton(onPressed: _index > 0 ? _prev : null, child: const Text('Prev')),
                const SizedBox(width: 12),
                ElevatedButton(
                  onPressed: _index < games.length - 1 ? _next : null,
                  child: const Text('Next'),
                ),
                const SizedBox(width: 12),
                ElevatedButton(onPressed: _buy, child: const Text('Buy')), // ⭐⭐⭐
              ],
            ),
            const SizedBox(height: 12),
            if (_cart.isNotEmpty) Text('Total: ${sum.toStringAsFixed(2)} USD'), // ⭐⭐⭐
          ],
        ),
      ),
    );
  }
}
