// Міні-проєкт II · Камінь–ножиці–папір
// Прізвище Імʼя · рівень: ⭐⭐
// Гра після уроку 14: крок 1, 2, 3 і три фінали

const moves = ['камінь', 'ножиці', 'папір'];

// Крок 1: хід компʼютера — випадковий елемент масиву
function computerMove() {
  return moves[Math.floor(Math.random() * moves.length)];
}

// Крок 1: чистий ввід — без пробілів по краях і без великих літер
function cleanMove(raw) {
  return raw.trim().toLowerCase();
}

// Крок 2: таблиця 3 × 3 у коді — нічия першою, далі три пари «гравець бʼє»
function whoWins(player, computer) {
  if (player === computer) return 'нічия';
  if ((player === 'камінь' && computer === 'ножиці') ||
      (player === 'ножиці' && computer === 'папір') ||
      (player === 'папір' && computer === 'камінь')) {
    return 'гравець';
  }
  return 'компʼютер';
}

// Крок 3: матч до трьох перемог
let playerWins = 0;
let computerWins = 0;
while (playerWins < 3 && computerWins < 3) {
  const raw = prompt(`Рахунок ${playerWins}:${computerWins}. Твій хід?`);
  if (raw === null) break;                 // Скасувати
  const player = cleanMove(raw);
  if (!moves.includes(player)) {
    alert('Такого ходу немає');
    continue;                              // раунд не зараховано
  }
  const computer = computerMove();
  const result = whoWins(player, computer);
  if (result === 'гравець') playerWins++;
  if (result === 'компʼютер') computerWins++;
  alert(`Компʼютер: ${computer}. Результат: ${result}`);
}

// Після циклу: хто дійшов до 3 — або гру перервано
if (playerWins === 3) {
  alert('Перемога в матчі!');
} else if (computerWins === 3) {
  alert('Матч виграв компʼютер');
} else {
  alert('Гру перервано');
}

knp/script.js · після уроку 15 
// Міні-проєкт II · Камінь–ножиці–папір
// Прізвище Імʼя · рівень: ⭐⭐⭐
// зроблено: крок 1–3 (урок 14), фінал, історія, реванш, к/н/п

const moves = ['камінь', 'ножиці', 'папір'];

// Хід компʼютера — випадковий елемент масиву
function computerMove() {
  return moves[Math.floor(Math.random() * moves.length)];
}

// Чистий ввід: без пробілів по краях і без великих літер
function cleanMove(raw) {
  return raw.trim().toLowerCase();
}

// НОВЕ · крок 6: скорочення к, н, п → повний хід
function fullMove(s) {
  if (s === 'к') return 'камінь';
  if (s === 'н') return 'ножиці';
  if (s === 'п') return 'папір';
  return s;
}

// Таблиця 3 × 3 у коді: нічия першою, далі три пари «гравець бʼє»
function whoWins(player, computer) {
  if (player === computer) return 'нічия';
  if ((player === 'камінь' && computer === 'ножиці') ||
      (player === 'ножиці' && computer === 'папір') ||
      (player === 'папір' && computer === 'камінь')) {
    return 'гравець';
  }
  return 'компʼютер';
}

// НОВЕ · крок 5: одна літера на раунд для історії
function letter(result) {
  if (result === 'гравець') return 'Г';
  if (result === 'компʼютер') return 'К';
  return '=';
}

// НОВЕ · крок 4: фінальне повідомлення — хто дійшов до 3
function finalText(p, c) {
  if (p === 3) return `Перемога в матчі! ${p}:${c}`;
  if (c === 3) return `Матч виграв компʼютер. ${p}:${c}`;
  return 'Гру перервано';
}

// НОВЕ · крок 6: реванш — do...while, рахунок обнуляється всередині
do {
  let playerWins = 0;
  let computerWins = 0;
  let draws = 0;
  const history = [];

  while (playerWins < 3 && computerWins < 3) {
    const raw = prompt(`Рахунок ${playerWins}:${computerWins}. Твій хід?`);
    if (raw === null) break;                 // Скасувати
    const player = fullMove(cleanMove(raw));
    if (!moves.includes(player)) {
      alert('Такого ходу немає');
      continue;                              // раунд не зараховано
    }
    const computer = computerMove();
    const result = whoWins(player, computer);
    if (result === 'гравець') playerWins++;
    if (result === 'компʼютер') computerWins++;
    if (result === 'нічия') draws++;
    history.push(letter(result));
    alert(`Компʼютер: ${computer}. Результат: ${result}`);
  }

  alert(`${finalText(playerWins, computerWins)}
Раундів: ${history.length}, нічиїх: ${draws}
Історія: ${history.join(' ')}`);
} while (confirm('Ще матч?'));
