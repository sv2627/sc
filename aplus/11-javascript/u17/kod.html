script.js · фінальний код уроку 
// Урок 17 · Прізвище Імʼя · рівень: ⭐⭐⭐

// ── Старт: сторінка після уроку 16 ──
const title = document.querySelector('#title');
title.textContent = 'GameStore · ігри для всіх';
const d = new Date();
document.querySelector('#today').textContent =
  `Сьогодні ${d.getDate()}.${d.getMonth() + 1}.${d.getFullYear()}`;
document.querySelector('.game').innerHTML = 'Minecraft — <b>649 грн</b>';
const hour = d.getHours();   // перевірка: тимчасово const hour = 20;
let hello = 'Доброго вечора';
if (hour < 12) {
  hello = 'Доброго ранку';
} else if (hour < 18) {
  hello = 'Доброго дня';
}
const note = document.querySelector('.note');
note.textContent = `${hello}! Ціни оновлено`;

// ── 1. style: одна властивість, назви без дефіса (camelCase), розміри з px ──
title.style.padding = '8px 12px';
const size = 32;
title.style.fontSize = `${size}px`;

// ── 2. classList: add / remove / toggle / contains / replace ──
const first = document.querySelector('.game');
first.classList.add('hit');              // назва класу — без крапки
const terra = document.querySelector('.sale');
terra.classList.replace('sale', 'old');  // знижка скінчилась
console.log(first.classList.contains('hit'));   // true

// ── 3. нічна тема: увечері (з 18) і вночі (до 7) ──
const night = hour >= 18 || hour < 7;
document.body.classList.toggle('dark', night);   // true — додати, false — прибрати
if (night) {
  note.textContent += ' 🌙 Нічна тема';
}

// ── 4. усі ігри: querySelectorAll — список, клас ставимо кожному в циклі ──
const games = document.querySelectorAll('.game');
const prices = [649, 319, 199];           // у тому ж порядку, що й ігри
let cheap = 0;
for (let i = 0; i < games.length; i++) {
  games[i].classList.add('card');
  if (prices[i] < 300) {
    games[i].classList.add('cheap');
    cheap++;
  }
}
note.textContent += ` · Ігор: ${games.length}, дешевих: ${cheap}`;
