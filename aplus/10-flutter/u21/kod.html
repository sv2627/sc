import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.deepPurple),
      home: const TicketScreen(),
    );
  }

class TicketScreen extends StatefulWidget {
  const TicketScreen({super.key});

  @override
  State<TicketScreen> createState() => _TicketScreenState();
}

class _TicketScreenState extends State<TicketScreen> {
  final TextEditingController _age = TextEditingController(); // що введено в поле
  bool _is3D = false; // галочка «3D»
  String _result = ''; // напис під кнопкою

  void _check() {
    int? age = int.tryParse(_age.text); // текст -> число; не число -> null
    setState(() {
      // 1. ДОПИШИ ТУТ: заміни рядок нижче на if / else if / else:

      //    age < 12    -> 'Child ticket: 5 USD'
      //    інакше      -> 'Adult ticket: 8 USD'
      _result = 'Age: $age';
    });
  }

  @override
  void dispose() {
    _age.dispose(); // контролер прибираємо разом з екраном
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Cinema'), centerTitle: true),
      body: Center(
        child: SizedBox(
          width: 320,
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              TextField(
                controller: _age,
                keyboardType: TextInputType.number,
                decoration: const InputDecoration(labelText: 'Age', border: OutlineInputBorder()),
              ),
              Row(
                children: [
                  Checkbox(
                    value: _is3D,
                    onChanged: (bool? v) {
                      setState(() {
                        _is3D = v ?? false;
                      });

                  ),

                ],
              ),
              ElevatedButton(onPressed: _check, child: const Text('Check')),
              const SizedBox(height: 16),

// Урок 21 · практика 1 · привітання: поле вводу, кнопка, напис.
// Текст з поля читаємо через TextEditingController, напис міняємо через setState.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      home: const HelloScreen(),
    );
  }
}

class HelloScreen extends StatefulWidget {
  const HelloScreen({super.key});

  @override
  State<HelloScreen> createState() => _HelloScreenState();
}

class _HelloScreenState extends State<HelloScreen> {
  final TextEditingController _name = TextEditingController();
  String _greeting = '';
  bool _error = false; // ⭐⭐ поле порожнє — напис червоний

  void _sayHello() {
    setState(() {
      // trim() прибирає пробіли по краях: '  ' -> ''
      if (_name.text.trim().isEmpty) {
        _greeting = 'Enter your name'; // ⭐⭐
        _error = true;
      } else {
        _greeting = 'Hello, ${_name.text.trim()}!';
        _error = false;
      }
    });
  }

  // ⭐⭐⭐ очистити поле і напис
  void _clear() {
    _name.clear();
    setState(() {
      _greeting = '';
      _error = false;
    });
  }

  @override
  void dispose() {
    _name.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Hello'), centerTitle: true),
      body: Center(
        child: SizedBox(
          width: 320, // TextField займає всю ширину — обмежуємо
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              TextField(
                controller: _name,
                decoration: const InputDecoration(labelText: 'Name', border: OutlineInputBorder()),
              ),
              const SizedBox(height: 12),
              Row(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  ElevatedButton(onPressed: _sayHello, child: const Text('Say hello')),
                  const SizedBox(width: 12),
                  OutlinedButton(onPressed: _clear, child: const Text('Clear')), // ⭐⭐⭐
                ],
              ),
              const SizedBox(height: 16),
              Text(
                _greeting,
                style: TextStyle(fontSize: 24, color: _error ? Colors.red : Colors.black),
              ),
            ],
          ),
        ),
      ),
    );
  }
}

// Урок 21 · зразок поточної роботи «Замовлення гри» · рівні ⭐–⭐⭐⭐.
// Поля вводу, Switch і Checkbox; результат і помилки — умовний рендер.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      home: const OrderScreen(),
    );
  }
}

class OrderScreen extends StatefulWidget {
  const OrderScreen({super.key});

  @override
  State<OrderScreen> createState() => _OrderScreenState();
}

class _OrderScreenState extends State<OrderScreen> {
  static const double price = 19.99; // ціна гри Celeste

  final TextEditingController _name = TextEditingController(); // ⭐
  final TextEditingController _qty = TextEditingController(); // ⭐⭐
  bool _gift = false; // ⭐ Switch: подарункова упаковка
  bool _student = false; // ⭐⭐⭐ Checkbox: знижка 10 %
  String _result = '';
  bool _error = false;

  void _order() {
    int? qty = int.tryParse(_qty.text); // ⭐⭐ '2' -> 2, 'two' -> null
    setState(() {
      if (qty == null || qty <= 0) {
        _result = 'Enter a number from 1';
        _error = true;
        return;
      }
      double total = qty * price;
      if (_gift) total += 2; // ⭐⭐⭐ упаковка +2 USD
      if (_student) total *= 0.9; // ⭐⭐⭐ −10 %
      _result = '${_name.text}: ${total.toStringAsFixed(2)} USD';
      _error = false;
    });
  }

  @override
  void dispose() {
    _name.dispose();
    _qty.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Order: Celeste'), centerTitle: true),
      body: Center(
        child: SizedBox(
          width: 340,
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              TextField(
                controller: _name,
                // ⭐⭐⭐ кожна літера -> build знову, щоб кнопка ожила
                onChanged: (String v) => setState(() {}),
                decoration: const InputDecoration(labelText: 'Name', border: OutlineInputBorder()),
              ),
              const SizedBox(height: 10),
              TextField(
                controller: _qty,
                keyboardType: TextInputType.number,
                decoration: const InputDecoration(labelText: 'Quantity', border: OutlineInputBorder()),
              ),
              Row(
                children: [
                  Switch(
                    value: _gift,
                    onChanged: (bool v) {
                      setState(() {
                        _gift = v;
                      });
                    },
                  ),
                  const Text('Gift wrap'),
                  const SizedBox(width: 12),
                  // ⭐⭐⭐ у Checkbox значення bool? — тому v ?? false
                  Checkbox(
                    value: _student,
                    onChanged: (bool? v) {
                      setState(() {
                        _student = v ?? false;
                      });
                    },
                  ),
                  const Text('Student'),
                ],
              ),
              if (_gift) const Text('🎁 +2 USD'), // ⭐ є лише тоді, коли Switch увімкнено
              const SizedBox(height: 8),
              ElevatedButton(
                // ⭐⭐⭐ поки імʼя порожнє — кнопка сіра
                onPressed: _name.text.isEmpty ? null : _order,
                child: const Text('Order'),
              ),
              const SizedBox(height: 14),
              if (_result.isNotEmpty)
                Text(
                  _result,
                  style: TextStyle(fontSize: 22, color: _error ? Colors.red : Colors.green.shade800),
                ),
            ],
          ),
        ),
      ),
    );
  }
}
