// DartPad: заміни весь код цим, натисни Run. Допиши два місця з позначкою ДОПИШИ ТУТ.
// Уже працює: вводиш назву фільму, натискаєш Add — фільм зʼявляється в списку.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.deepPurple),
      home: const MovieScreen(),
    );
  }
}

class MovieScreen extends StatefulWidget {
  const MovieScreen({super.key});

  @override
  State<MovieScreen> createState() => _MovieScreenState();
}

class _MovieScreenState extends State<MovieScreen> {
  final TextEditingController _movie = TextEditingController(); // що введено в поле
  final List<String> _movies = ['Inception', 'Up']; // список фільмів

  void _add() {
    String text = _movie.text.trim();
    if (text.isEmpty) return; // порожнє не додаємо
    setState(() {
      _movies.add(text);
    });
    _movie.clear();
  }

  @override
  void dispose() {
    _movie.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      // 1. ДОПИШИ ТУТ: заміни 'Movies' на 'Movies: N', де N — кількість фільмів у списку
      //    підказка: '... ${_movies.length}'
      appBar: AppBar(title: const Text('Movies')),
      body: Column(
        children: [
          Padding(
            padding: const EdgeInsets.all(12),
            child: Row(
              children: [
                Expanded(
                  child: TextField(
                    controller: _movie,
                    decoration: const InputDecoration(labelText: 'Movie', border: OutlineInputBorder()),
                  ),
                ),
                const SizedBox(width: 8),
                ElevatedButton(onPressed: _add, child: const Text('Add')),
              ],
            ),
          ),
          Expanded(
            child: ListView.builder(
              itemCount: _movies.length,
              itemBuilder: (BuildContext context, int i) {
                return ListTile(
                  leading: const Icon(Icons.movie),
                  title: Text(_movies[i]),
                  // 2. ДОПИШИ ТУТ: кнопка-кошик праворуч, що видаляє цей фільм
                  //    trailing: IconButton(icon: const Icon(Icons.delete), onPressed: () { ... }),
                  //    усередині — setState(() { _movies.removeAt(i); });
                );
              },
            ),
          ),
        ],
      ),
    );
  }
}

// Урок 23 · практика 1 · список покупок: поле, кнопка Add, ListView — фінальний вигляд.
// DartPad: заміни весь код цим і натисни Run. VS Code: заміни весь lib/main.dart, Ctrl+S, hot restart.
// Головне: список — поле стану, додаємо через setState, ListView.builder сам показує новий рядок.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.teal),
      home: const ShoppingScreen(),
    );
  }
}

class ShoppingScreen extends StatefulWidget {
  const ShoppingScreen({super.key});

  @override
  State<ShoppingScreen> createState() => _ShoppingScreenState();
}

class _ShoppingScreenState extends State<ShoppingScreen> {
  final TextEditingController _item = TextEditingController(); // що введено в поле
  final List<String> _items = ['Milk', 'Bread']; // список змінюється -> без const

  void _add() {
    String text = _item.text.trim(); // прибрали пробіли по краях
    if (text.isEmpty) return; // ⭐⭐ порожнє не додаємо
    setState(() {
      _items.add(text); // додали в кінець списку
    });
    _item.clear(); // поле знову порожнє
  }

  @override
  void dispose() {
    _item.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('Shopping: ${_items.length}')), // лічильник у шапці
      body: Column(
        children: [
          Padding(
            padding: const EdgeInsets.all(12),
            child: Row(
              children: [
                Expanded( // TextField у Row — тільки в Expanded
                  child: TextField(
                    controller: _item,
                    decoration: const InputDecoration(labelText: 'Item', border: OutlineInputBorder()),
                  ),
                ),
                const SizedBox(width: 8),
                ElevatedButton(onPressed: _add, child: const Text('Add')),
              ],
            ),
          ),
          Expanded( // ListView у Column — теж тільки в Expanded
            child: ListView.builder(
              itemCount: _items.length,
              itemBuilder: (BuildContext context, int i) {
                return ListTile(
                  leading: const Icon(Icons.shopping_cart),
                  title: Text(_items[i]),
                  // ⭐⭐⭐ кошик праворуч — видалити саме цей рядок
                  trailing: IconButton(
                    icon: const Icon(Icons.delete),
                    onPressed: () {
                      setState(() {
                        _items.removeAt(i);
                      });
                    },
                  ),
                );
              },
            ),
          ),
        ],
      ),
    );
  }
}

// Урок 23 · Практична № 4 «Список із додаванням» · зразок ⭐–⭐⭐⭐.
// DartPad: заміни весь код цим і натисни Run. VS Code: заміни весь lib/main.dart, Ctrl+S, hot restart.
// ⭐ ListView.builder + кнопка · ⭐⭐ TextField + видалення · ⭐⭐⭐ список обʼєктів, Checkbox, лічильник в AppBar.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

// ⭐⭐⭐ Завдання — сутність -> class: назва і чи зроблено.
class Task {
  final String title;
  bool done; // змінюється від галочки -> не final

  Task(this.title, {this.done = false});
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      home: const TaskScreen(),
    );
  }
}

class TaskScreen extends StatefulWidget {
  const TaskScreen({super.key});

  @override
  State<TaskScreen> createState() => _TaskScreenState();
}

class _TaskScreenState extends State<TaskScreen> {
  final TextEditingController _title = TextEditingController();
  final List<Task> _tasks = [Task('Read chapter 3'), Task('Buy a notebook', done: true)];

  // ⭐⭐⭐ скільки зроблено — рахуємо циклом
  int _doneCount() {
    int n = 0;
    for (Task t in _tasks) {
      if (t.done) n++;
    }
    return n;
  }

  void _add() {
    String text = _title.text.trim();
    if (text.isEmpty) return; // порожнє не додаємо
    setState(() {
      _tasks.add(Task(text));
    });
    _title.clear();
  }

  @override
  void dispose() {
    _title.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      // ⭐⭐⭐ лічильник у шапці: зроблено / усього
      appBar: AppBar(title: Text('Tasks: ${_doneCount()} / ${_tasks.length}')),
      body: Column(
        children: [
          Padding(
            padding: const EdgeInsets.all(12),
            child: Row(
              children: [
                // ⭐⭐ поле для назви
                Expanded(
                  child: TextField(
                    controller: _title,
                    decoration: const InputDecoration(labelText: 'New task', border: OutlineInputBorder()),
                  ),
                ),
                const SizedBox(width: 8),
                // ⭐ кнопка додає рядок
                ElevatedButton(onPressed: _add, child: const Text('Add')),
              ],
            ),
          ),
          if (_tasks.isEmpty) const Text('No tasks yet'), // порожній список — підказка
          Expanded(
            // ⭐ список, що гортається
            child: ListView.builder(
              itemCount: _tasks.length,
              itemBuilder: (BuildContext context, int i) {
                Task t = _tasks[i];
                return Card(
                  child: ListTile(
                    // ⭐⭐⭐ галочка: зроблено чи ні
                    leading: Checkbox(
                      value: t.done,
                      onChanged: (bool? v) {
                        setState(() {
                          t.done = v ?? false;
                        });
                      },
                    ),
                    title: Text(
                      t.title,
                      style: TextStyle(decoration: t.done ? TextDecoration.lineThrough : null),
                    ),
                    // ⭐⭐ видалити саме цей рядок
                    trailing: IconButton(
                      icon: const Icon(Icons.delete),
                      onPressed: () {
                        setState(() {
                          _tasks.removeAt(i);
                        });
                      },
                    ),
                  ),
                );
              },
            ),
          ),
        ],
      ),
    );
  }
}
