script.js · фінальний код уроку 
// Урок 16 · Прізвище Імʼя · рівень: ⭐⭐

// 1. Знайти елемент за id і змінити текст
const title = document.querySelector('#title');
title.textContent = 'GameStore · ігри для всіх';

// 2. Сьогоднішня дата. Місяці рахуються з 0, тому + 1
const d = new Date();
const today = document.querySelector('#today');
today.textContent = `Сьогодні ${d.getDate()}.${d.getMonth() + 1}.${d.getFullYear()}`;

// 3. querySelector('.game') знаходить ПЕРШИЙ елемент із класом game
const first = document.querySelector('.game');
first.innerHTML = 'Minecraft — <b>649 грн</b>';   // innerHTML — з тегами

// 4. Привітання за годиною
const hour = d.getHours();
let hello = 'Доброго вечора';
if (hour < 12) {
  hello = 'Доброго ранку';
} else if (hour < 18) {
  hello = 'Доброго дня';
}
document.querySelector('.note').textContent = `${hello}! Ціни оновлено`;
