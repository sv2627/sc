1 · Що поверне?

function beats(a, b) {
  return (a === 'камінь' && b === 'ножиці')
    || (a === 'ножиці' && b === 'папір')
    || (a === 'папір' && b === 'камінь');
}
function whoWins(p, c) {
  if (p === c) return 'нічия';
  if (beats(p, c)) return 'гравець';
  return 'компʼютер';
}
