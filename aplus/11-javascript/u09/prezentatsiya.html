script.js
const cart = ['Tetris', 'Doom']; cart.push('Portal'); console.log(cart); // ['Tetris', 'Doom', 'Portal'] const last = cart.pop(); console.log(last); // Portal console.log(cart); // ['Tetris', 'Doom']

join

Масив → один рядок для alert

script.js
const cart = ['Tetris', 'Doom', 'Portal']; cart.join(', '); // 'Tetris, Doom, Portal' cart.join(' + '); // 'Tetris + Doom + Portal' alert(`Кошик: ${cart}`); // Кошик: Tetris,Doom,Portal
join(', ') — склеює елементи, між ними ставить те, що в дужках
Без join масив у рядку склеюється комами без пробілів
Масив при цьому не змінюється — join дає новий рядок

Вибери

Коли for, а коли for...of

for...of — масив не зміниться
const prices = [500, 1000, 200]; for (let p of prices) { p = p * 0.9; } // [500, 1000, 200]
for з індексом — змінює сам масив
const prices = [500, 1000, 200]; for (let i = 0; i < prices.length; i++) { prices[i] = prices[i] * 0.9; } // [450, 900, 180]
for...of — лише прочитати кожен: сума, пошук, вивід
for — потрібен номер або треба змінити елементи

Сума і середня ціна

script.js
const prices = [499, 1299, 250, 899]; let sum = 0; for (const p of prices) { sum += p; } const avg = sum / prices.length; console.log(`Разом: ${sum} грн`); console.log(`Середня: ${avg} грн`);
КонсольРазом: 2947 грн Середня: 736.75 грн
Накопичувач з уроку 8: let sum = 0 до циклу
Середнє = сума ÷ length
Порожній масив: 0 / 0 → NaN

Виправ

Має бути середнє 10. Що не так?

script.js · зламано навмисно
const marks = [10, 8, 12]; let sum; for (let i = 1; i <= marks.length; i++) { sum += marks[i]; } console.log(sum / marks.length);
КонсольNaN

Три біди, червоного рядка немає

Знайди всі три, потім клацни

1. let sum; — без нуля: undefined + 8 = NaN

2. i = 1 — перша оцінка пропущена. Старт з 0

3. i <= marks.length — читає marks[3], тобто undefined. Потрібно <

Полагоджене дає 10. Або коротше — for (const m of marks): ні старту, ні межі.
