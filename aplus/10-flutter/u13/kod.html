// Домашнє · урок 13 · «Радіо». Прізвище: ...
// Кнопка Next уже працює. Допиши два місця з позначкою ДОПИШИ ТУТ.
// Перевір: тричі Next -> Station 4 / 4, Rock FM, Next сіра;
//          ще чотири рази Prev -> Station 1 / 4, Jazz FM, Prev сіра.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.teal),
      home: const RadioPage(),
    );
  }
}

class RadioPage extends StatefulWidget {
  const RadioPage({super.key});

  @override
  State<RadioPage> createState() => _RadioPageState();
}

class _RadioPageState extends State<RadioPage> {
  final List<String> _stations = ['Jazz FM', 'News 24', 'Pop Hits', 'Rock FM'];
  int _index = 0; // номер станції у списку, з нуля

  // Готово: наступна, але не далі останньої.
  void _next() {
    setState(() {
      if (_index < _stations.length - 1) {
        _index++;
      }
    });
  }

  // 1. ДОПИШИ ТУТ: попередня, але не раніше першої (номер 0).
  //    Так само, як _next вище: setState, усередині if (_index > 0) і _index--.
  void _prev() {
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      // 2. ДОПИШИ ТУТ: замість 'Radio' — 'Station 1 / 4'.
      //    Номер для людини з одиниці: ${_index + 1}, а скільки всього — ${_stations.length}.
  //    const перед Text прибери — у тексті тепер змінна.
      appBar: AppBar(title: const Text('Radio')),
      body: Center(
        child: Column(
          mainAxisAlignment: .center,
          children: [
            Text(
              _stations[_index],
              style: Theme.of(context).textTheme.displaySmall,
            ),
            Row(
              mainAxisAlignment: .center,
              children: [
                ElevatedButton(
                  onPressed: _index > 0 ? _prev : null,
                  child: const Text('Prev'),
                ),
                ElevatedButton(
                  onPressed: _index < _stations.length - 1 ? _next : null,
                  child: const Text('Next'),
                ),
              ],
            ),
          ],
        ),
      ),
    );
  }
}

// Урок 13 · 1. Базовий лічильник — той самий, що dartpad.dev/?sample=counter.
// Одна кругла кнопка «+», число росте. Від нього стартуємо.
import 'package:flutter/material.dart';

// Точка входу. runApp показує на екрані головний віджет — MyApp.
void main() => runApp(const MyApp());

// MyApp — рамка застосунку: назва, колір, перший екран.
// StatelessWidget — сам ніколи не змінюється.
class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Counter',
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.blue),
      home: const MyHomePage(title: 'Counter'),
    );
  }
}

// MyHomePage — екран, який змінюється. StatefulWidget — віджет зі станом,
// а сам стан живе в класі _MyHomePageState нижче.
class MyHomePage extends StatefulWidget {
  final String title;

  const MyHomePage({super.key, required this.title});

  @override
  State<MyHomePage> createState() => _MyHomePageState();
}

class _MyHomePageState extends State<MyHomePage> {
  int _counter = 0; // стан: число на екрані

  // setState — «дані змінились, перемалюй екран».
  void _incrementCounter() {
    setState(() {
      _counter++;
    });
  }

  @override
  Widget build(BuildContext context) {
    // Scaffold — каркас сторінки: шапка, тіло, кругла кнопка.
    return Scaffold(
      appBar: AppBar(title: Text(widget.title)),
      body: Center(
        // Column — віджети один під одним, .center — по центру.
        child: Column(
          mainAxisAlignment: .center,
          children: [
            const Text('You have pushed the button this many times:'),
            Text(
              '$_counter', // число стає текстом через $
              style: Theme.of(context).textTheme.headlineMedium,
            ),
          ],
        ),
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: _incrementCounter, // без () — передаємо метод, а не викликаємо
        tooltip: 'Increment',
        child: const Icon(Icons.add),
      ),
    );
  }
}
2 · Лічильник з + і − (практика 1) 
// Урок 13 · 2. Модифікований лічильник: кнопки «−» і «+» поруч, не нижче 0,
// кругла кнопка тепер скидає до 0. Усе, що змінилось від базового, позначено НОВЕ.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Counter',
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.blue),
      home: const MyHomePage(title: 'Counter: + and -'),
    );
  }
}

class MyHomePage extends StatefulWidget {
  final String title;

  const MyHomePage({super.key, required this.title});

  @override
  State<MyHomePage> createState() => _MyHomePageState();
}

class _MyHomePageState extends State<MyHomePage> {
  int _counter = 0;

  // «+» — як і було.
  void _increment() {
    setState(() {
      _counter++;
    });
  }

  // НОВЕ: «−». if не пускає нижче нуля.
  void _decrement() {
    setState(() {
      if (_counter > 0) {
        _counter--;
      }
    });
  }

  // НОВЕ: скинути до нуля.
  void _reset() {
    setState(() {
      _counter = 0;
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text(widget.title)),
      body: Center(
        child: Column(
          mainAxisAlignment: .center,
          children: [
            const Text('Count:'),
            Text(
              '$_counter',
              style: Theme.of(context).textTheme.displayMedium, // НОВЕ: ще більший шрифт
            ),
            // НОВЕ: Row — дві кнопки поруч під числом.
            Row(
              mainAxisAlignment: .center,
              children: [
                ElevatedButton(
                  // На нулі кнопка сіра: null — «натискати нічого».
                  onPressed: _counter > 0 ? _decrement : null,
                  child: const Icon(Icons.remove),
                ),
                ElevatedButton(
                  onPressed: _increment,
                  child: const Icon(Icons.add),
                ),
              ],
            ),
          ],
        ),
      ),
      // НОВЕ: кругла кнопка тепер не «+», а «скинути».
      floatingActionButton: FloatingActionButton(
        onPressed: _reset,
        tooltip: 'Reset',
        child: const Icon(Icons.refresh),
      ),
    );
  }
}
3 · Плейлист ⭐–⭐⭐⭐ (практика 2) 
// Урок 13 · Практика 2 «Плейлист» — фінальна версія ⭐–⭐⭐⭐.
// Той самий лічильник, тільки число — це номер відео у списку.
// «−» — попереднє, «+» — наступне. Плеєра ще немає: замість нього — назва.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

// Відео — сутність (назва і тривалість) -> class.
class Video {
  final String title;
  final int minutes;
  const Video(this.title, this.minutes);
}

// Порядок важливий, ходимо за номером -> List.
const List<Video> videos = [
  Video('Flutter in 100 seconds', 2),
  Video('Expanded', 1),
  Video('Spacer', 1),
  Video('Wrap', 1),
];

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.red),
      home: const PlaylistPage(),
    );
  }
}

class PlaylistPage extends StatefulWidget {
  const PlaylistPage({super.key});

  @override
  State<PlaylistPage> createState() => _PlaylistPageState();
}

class _PlaylistPageState extends State<PlaylistPage> {
  int _index = 0; // номер відео у списку, з нуля

  // ⭐ Наступне — не далі останнього.
  void _next() {
    setState(() {
      if (_index < videos.length - 1) {
        _index++;
      }
    });
  }

  // ⭐ Попереднє — не раніше першого.
  void _prev() {
    setState(() {
      if (_index > 0) {
        _index--;
      }
    });
  }

  // ⭐⭐⭐ По колу: після останнього — знову перше. % — остача від ділення.
  void _loopNext() {
    setState(() {
      _index = (_index + 1) % videos.length;
    });
  }

  // ⭐⭐⭐ Скільки хвилин у всьому плейлисті.
  int _totalMinutes() {
    int sum = 0;
    for (Video v in videos) {
      sum += v.minutes;
    }
    return sum;
  }

  @override
  Widget build(BuildContext context) {
    Video v = videos[_index]; // поточне відео
    return Scaffold(
      // ⭐ Номер для людини — з одиниці, тому _index + 1.
      appBar: AppBar(title: Text('Video ${_index + 1} / ${videos.length}')),
      body: Center(
        child: Column(
          mainAxisAlignment: .center,
          children: [
            Text(v.title, style: Theme.of(context).textTheme.headlineMedium),
            Text('${v.minutes} min'),
            Row(
              mainAxisAlignment: .center,
              children: [
                // ⭐⭐ На краях кнопка сіра.
                ElevatedButton(
                  onPressed: _index > 0 ? _prev : null,
                  child: const Icon(Icons.skip_previous),
                ),
                ElevatedButton(
                  onPressed: _index < videos.length - 1 ? _next : null,
                  child: const Icon(Icons.skip_next),
                ),
              ],
            ),
            // ⭐⭐ Останнє відео — напис.
            Text(_index == videos.length - 1 ? 'Last video' : ''),
            Text('Playlist: ${_totalMinutes()} min'),
          ],
        ),
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: _loopNext,
        tooltip: 'Loop',
        child: const Icon(Icons.repeat),
      ),
    );
  }
}

// Урок 13 · YouTube-плеєр на Flutter — перший застосунок із пакетом.
// Це «Плейлист» з практики 2, тільки замість назви — справжній плеєр.
// Пакет: youtube_player_flutter 10 (pub.dev). У DartPad не запуститься —
// потрібен проєкт: flutter create -> flutter pub add youtube_player_flutter.
// Запуск: flutter run -d chrome · збірка сайту: flutter build web
import 'package:flutter/material.dart';
// НОВЕ: імпорт пакета — після flutter pub add він є в pubspec.yaml.
import 'package:youtube_player_flutter/youtube_player_flutter.dart';

void main() => runApp(const MyApp());

// Відео — сутність: id з посилання YouTube і назва -> class.
// id — це те, що в посиланні після watch?v=
class Video {
  final String id;
  final String title;
  const Video(this.id, this.title);
}

// Порядок важливий, ходимо за номером -> List.
const List<Video> videos = [
  Video('lHhRhPV--G0', 'Flutter in 100 seconds'),
  Video('_rnZaagadyo', 'Expanded · Widget of the Week'),
  Video('7FJgd7QN1zI', 'Spacer · Widget of the Week'),
  Video('z5iw2SeFx2M', 'Wrap · Widget of the Week'),
];

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Flutter videos',
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.red),
      home: const PlayerPage(),
    );
  }
}

class PlayerPage extends StatefulWidget {
  const PlayerPage({super.key});

  @override
  State<PlayerPage> createState() => _PlayerPageState();
}

class _PlayerPageState extends State<PlayerPage> {
  int _index = 0; // номер відео у списку, з нуля — як _counter

  // НОВЕ: пульт плеєра. late — «значення буде, але трохи пізніше» (в initState).
  late final YoutubePlayerController _controller;

  // НОВЕ: initState — один раз, коли екран зʼявляється. Тут створюємо пульт.
  @override
  void initState() {
    super.initState();
    _controller = YoutubePlayerController.fromVideoId(
      videoId: videos[_index].id,
      autoPlay: false, // браузер не дає вмикати звук без кліку
    );
  }

  // НОВЕ: dispose — коли екран зникає. Пульт закриваємо, щоб не грав «у фоні».
  @override
  void dispose() {
    _controller.close();
    super.dispose();
  }

  // Показати відео з номером i: змінити стан і сказати плеєру, що вантажити.
  void _show(int i) {
    setState(() {
      _index = i;
    });
    _controller.loadVideoById(videoId: videos[i].id);
  }

  // «+» — наступне, «−» — попереднє. Межі — як у лічильнику «не нижче 0».
  void _next() {
    if (_index < videos.length - 1) _show(_index + 1);
  }

  void _prev() {
    if (_index > 0) _show(_index - 1);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('Video ${_index + 1} / ${videos.length}')),
      body: Center(
        // Щоб на широкому екрані плеєр не розтягувався на всю ширину.
        child: ConstrainedBox(
          constraints: const BoxConstraints(maxWidth: 800),
          child: Column(
            mainAxisAlignment: .center,
            children: [
              // НОВЕ: сам плеєр — віджет із пакета. Йому досить пульта.
              YoutubePlayer(controller: _controller),
              Padding(
                padding: const EdgeInsets.all(16),
                child: Text(
                  videos[_index].title,
                  style: Theme.of(context).textTheme.titleLarge,
                ),
              ),
              Row(
                mainAxisAlignment: .center,
                children: [
                  ElevatedButton(
                    onPressed: _index > 0 ? _prev : null,
                    child: const Icon(Icons.skip_previous),
                  ),
                  ElevatedButton(
                    onPressed: _index < videos.length - 1 ? _next : null,
                    child: const Icon(Icons.skip_next),
                  ),
                ],
              ),
            ],
          ),
        ),
      ),
    );
  }
}
