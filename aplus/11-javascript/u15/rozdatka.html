function finalText(p, c) {
  if (p === 3) return `Перемога ${p}:${c}`;
  if (c === 3) return `Програш ${p}:${c}`;
  return 'Гру перервано';
}
function letter(result) {
  if (result === 'гравець') return 'Г';
  if (result === 'компʼютер') return 'К';
  return '=';
}

const history = [];
do {
  let p = 0;
  let c = 0;
  while (p < 3 || c < 3) {
    const raw = prompt('Твій хід?');
    if (raw === null) break;
    const pl = raw.trim().toLowerCase();
    if (moves.includes(pl)) continue;
    const r = whoWins(pl, computerMove());
    if (r === 'гравець') p++;
