3 · style.css — нові правила уроку 
/* Розминка: картка — якір, бейдж у правому верхньому куті */
.card {
  position: relative;
  transition: all 0.3s;      /* плавно все, що міняє :hover — і туди, і назад */
}
.badge {
  position: absolute;
  top: 8px;
  right: 8px;
  background: #af3b41;
  color: white;
  padding: 0.2rem 0.5rem;
}

/* Крок 1–3: миша над карткою — фон, підйом і збільшення */
.card:hover {
  background-color: #f6eef0;
  transform: translateY(-6px) scale(1.05);   /* дві функції — в ОДНОМУ рядку */
  box-shadow: 0 8px 16px gray;
}

/* Кнопка: плавно колір і підйом */
.btn {
  transition: background-color 0.3s, color 0.3s, transform 0.3s;
}
.btn:hover {
  background-color: #591429;
  color: white;
  transform: translateY(-3px);   /* мінус — вгору */
}
.btn:active {
  transform: scale(0.95);        /* ⭐⭐ поки тримаєш — «вдавлюється» */
}

/* Крок 4: липка шапка */
header {
  position: sticky;
  top: 0;                /* без top не прилипає */
  z-index: 10;           /* вище за картки, що збільшуються */
  background: #e2e7f6;   /* без фону картки просвічують */
}

/* ⭐⭐⭐ Кнопка «вгору»: <a href="#top" class="up">↑</a> в кінці body, у header — id="top" */
.up {
  position: fixed;
  bottom: 1rem;
  right: 1rem;

  background: #0253b3;
  color: white;
  text-decoration: none;
  transition: transform 0.3s;
}
