Компʼютер: ${computerMove()}`); }

Функція whoWins: таблиця 3 × 3 у коді

js-11/knp/script.js
function whoWins(player, computer) { if (player === computer) return 'нічия'; if ((player === 'камінь' && computer === 'ножиці') || (player === 'ножиці' && computer === 'папір') || (player === 'папір' && computer === 'камінь')) { return 'гравець'; } return 'компʼютер'; }
Три клітинки «Г» з таблиці — три пари в дужках через ||
Нічия — першою: вужча умова першою
Решта шість клітинок мінус три нічиї — це «К». Окремо їх не перелічуємо

Матч до трьох перемог

js-11/knp/script.js · замість одного раунду
let playerWins = 0; let computerWins = 0; while (playerWins < 3 && computerWins < 3) { const raw = prompt( `Рахунок ${playerWins}:${computerWins}. Твій хід?`); if (raw === null) break; const player = cleanMove(raw); if (!moves.includes(player)) { alert('Такого ходу немає'); continue; } const computer = computerMove(); const result = whoWins(player, computer); if (result === 'гравець') playerWins++; if (result === 'компʼютер') computerWins++; alert(`Компʼютер: ${computer}. Результат: ${result}`); }
&& — гра йде, поки обидва менше трьох
Нічия не змінює жодного лічильника — раунд просто повторюється
Без ++ цикл не закінчиться ніколи: рахунок має рости
Після циклу — хто дійшов до 3? Або break — «Гру перервано»
Практика 3 Матч до трьох · ⭐⭐10 хвилин
Приклад — після циклу
if (playerWins === 3) { alert('Перемога в матчі!'); } else if (computerWins === 3) { alert('Матч виграв компʼютер'); } else { alert('Гру перервано'); }
Тепер самостійно
  • Заміни один раунд циклом з кроку 3. Рахунок видно в кожному prompt
  • Після циклу — три фінали, як у прикладі. У фіналі — рахунок 3:1
  • Зіграй матч двічі до кінця. Нічия рахунок не змінює?
  • 'вогонь' посеред матчу — «Такого ходу немає», раунд не зараховано
  • ⭐⭐ Лічильники draws і rounds — у фінальному alert: «Раундів: 6, нічиїх: 2»
  • ⭐⭐⭐ Історія: масив history, після раунду push(`${player}–${computer}`), у кінці join(', ')

Матч, який не закінчується вчасно

script.js · зламано навмисно
let playerWins = 0; let computerWins = 0; while (playerWins < 3 || computerWins < 3) { const raw = prompt('Твій хід?'); if (raw === null) break; const player = cleanMove(raw); const computer = moves[Math.floor(Math.random() * 3) + 1]; const result = whoWins(player, computer); if (result === 'гравець') playerWins + 1; if (result === 'компʼютер') computerWins++; }

Три біди

Знайди всі три, потім клацни

1. || — матч іде, поки обидва не дійдуть до 3. Треба &&

2. + 1 в індексі — «камінь» не ходить ніколи, а раз на три — undefined

3. playerWins + 1 рахує, але не зберігає — перемоги гравця не ростуть. Треба playerWins++
