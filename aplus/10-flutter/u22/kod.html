Каталог у ListView ⭐–⭐⭐⭐ 
// Урок 22 · практика · каталог ігор у ListView ⭐–⭐⭐⭐ — фінальний вигляд.
// DartPad: заміни весь код цим і натисни Run. VS Code: заміни весь lib/main.dart, Ctrl+S, hot restart.
// ⭐ ListView.builder + Card + ListTile · ⭐⭐ номер у колі, дешеві ігри зеленим · ⭐⭐⭐ клік вибирає гру.
import 'package:flutter/material.dart';

void main() => runApp(const MyApp());

class Game {
  final String title;
  final String genre;
  final double price;

  const Game(this.title, this.genre, this.price);

  bool isCheap() => price < 15; // дешева — менше 15 USD
}

const List<Game> games = [
  Game('Stardew Valley', 'Farming sim', 14.99),
  Game('Celeste', 'Platformer', 19.99),
  Game('Terraria', 'Sandbox', 9.99),
  Game('Hades', 'Roguelike', 24.99),
  Game('Minecraft', 'Sandbox', 29.99),
  Game('Hollow Knight', 'Metroidvania', 14.99),
  Game('Portal 2', 'Puzzle', 9.99),
  Game('Cuphead', 'Run and gun', 19.99),
];

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: Colors.indigo),
      home: const GameList(), // ⭐⭐⭐ екран зі станом: памʼятає, яку гру вибрали
    );
  }
}

class GameList extends StatefulWidget {
  const GameList({super.key});

  @override
  State<GameList> createState() => _GameListState();
}

class _GameListState extends State<GameList> {
  int? _selected; // номер вибраної гри; null — ще нічого не вибрали

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        // ⭐⭐⭐ у шапці — назва вибраної гри
        title: Text(_selected == null ? 'Game store' : 'Selected: ${games[_selected!].title}'),
        centerTitle: true,
      ),
      // ⭐ ListView.builder будує рядки, лише коли вони видно на екрані
      body: ListView.builder(
        padding: const EdgeInsets.all(12),
        itemCount: games.length, // скільки рядків
        itemBuilder: (BuildContext context, int i) {
          Game g = games[i]; // гра для рядка номер i
          return Card(
            child: ListTile(
              // ⭐⭐ номер для людини — з 1, а не з 0
              leading: CircleAvatar(child: Text('${i + 1}')),
              title: Text(g.title),
              subtitle: Text(g.genre),
              trailing: Text(
                '${g.price} USD',
                // ⭐⭐ дешеві — зеленим
                style: TextStyle(
                  fontWeight: FontWeight.bold,
                  color: g.isCheap() ? Colors.green.shade800 : Colors.black,
                ),
              ),
              selected: i == _selected, // ⭐⭐⭐ вибраний рядок підсвічено
              onTap: () {
                setState(() {
                  _selected = i; // ⭐⭐⭐ запамʼятали номер — Flutter перемалював
                });
              },
            ),
          );
        },
      ),
    );
  }
}
