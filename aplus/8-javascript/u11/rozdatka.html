2 · Простеж max

function max(a) {
  let m = a[0];
  for (let i = 1; i < a.length; i++) {
    if (a[i] > m) m = a[i];
  }
  return m;
}

3 · Що вийде?

function pick(a) {
  const res = [];
  for (let i = 0; i < a.length; i++) {
    if (a[i] >= 8) res.push(a[i]);
  }
  return res;
}
