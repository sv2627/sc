// Урок 9 · Міні-проєкт I: симуляція бою — фінальна версія з уроку.
// Відкрий dartpad.dev, встав увесь код і натисни Run.
// ⭐ бій двох бійців · ⭐⭐ турнір і таблиця перемог · ⭐⭐⭐ крит, лікування, статистика.

import 'dart:math';

// Статистика всього турніру: тип події -> скільки разів (⭐⭐⭐).
Map<String, int> stats = {'hits': 0, 'crits': 0, 'heals': 0};

class Fighter {
  String name; // імʼя бійця
  int hp; // здоровʼя зараз
  int attack; // сила удару
  int maxHp = 0; // здоровʼя на початку — щоб відновити перед новим боєм
  int potions = 1; // лікувань на один бій (⭐⭐⭐)
  int damageDone = 0; // скільки шкоди завдав за турнір (⭐⭐⭐)

  // Конструктор: запамʼятовуємо початкове hp у maxHp.
  Fighter(this.name, this.hp, this.attack) {
    maxHp = hp;
  }

  // Живий, поки hp більше нуля.
  bool isAlive() => hp > 0;

  String info() => '$name: hp $hp, attack $attack';

  // Удар: сила + випадкове число від 0 до 5.
  void hit(Fighter other) {
    int damage = attack + Random().nextInt(6);
    // Шанс 1 з 5 на критичний удар — шкода ×2.
    if (Random().nextInt(5) == 0) {
      damage = damage * 2;
      print('CRIT!');
      stats.update('crits', (v) => v + 1);
    }
    other.hp = other.hp - damage;
    if (other.hp < 0) other.hp = 0; // hp не буває мінусовим
    damageDone = damageDone + damage;
    stats.update('hits', (v) => v + 1);
    print('$name hits ${other.name} for $damage. ${other.name}: ${other.hp} hp');
  }

  // Лікування: +20 hp, але не більше за maxHp.
  void heal() {
    hp = hp + 20;
    if (hp > maxHp) hp = maxHp;
    potions--;
    stats.update('heals', (v) => v + 1);
    print('$name heals. $name: $hp hp');
  }

  // Хід бійця: якщо hp менше третини і є зілля — лікується, інакше бʼє.
  void turn(Fighter other) {
    if (hp < maxHp ~/ 3 && potions > 0) {
      heal();
    } else {
      hit(other);
    }
  }

  // Перед кожним новим боєм — повне здоровʼя і одне зілля.
  void restore() {
    hp = maxHp;
    potions = 1;
  }
}

// Бій до переможця. Повертає того, хто вижив.
Fighter fight(Fighter a, Fighter b) {
  int round = 0;
  while (a.isAlive() && b.isAlive()) {
    round++;
    print('--- Round $round ---');
    a.turn(b);
    if (b.isAlive()) b.turn(a); // мертвий не відповідає
  }
  Fighter winner = a.isAlive() ? a : b;
  print('Winner: ${winner.name} in $round rounds, ${winner.hp} hp left');
  return winner;
}

void main() {
  List<Fighter> fighters = [
    Fighter('Andrii', 60, 10),
    Fighter('Bohdan', 70, 8),
    Fighter('Olha', 55, 11),
    Fighter('Daryna', 65, 9),
  ];

  // Таблиця перемог: спершу всім 0.
  Map<String, int> wins = {};
  for (Fighter f in fighters) wins[f.name] = 0;

  // Турнір «кожен з кожним»: j починається з i + 1,
  // щоб пара не билась двічі і боєць не бився сам із собою.
  for (int i = 0; i < fighters.length; i++) {
    for (int j = i + 1; j < fighters.length; j++) {
      fighters[i].restore();
      fighters[j].restore();
      Fighter w = fight(fighters[i], fighters[j]);
      wins.update(w.name, (v) => v + 1);
    }
  }

  // Таблиця перемог.
  print('=== Wins ===');
  wins.forEach((name, w) => print('$name: $w'));

  // Чемпіон — той, у кого найбільше перемог.
  String champion = '';
  int best = -1;
  wins.forEach((name, w) {
    if (w > best) {
      best = w;
      champion = name;
    }
  });
  print('Champion: $champion');

  // Статистика турніру (⭐⭐⭐).
  print('=== Stats ===');
  stats.forEach((k, v) => print('$k: $v'));
  for (Fighter f in fighters) {
    print('${f.name}: damage ${f.damageDone}');
  }
}
