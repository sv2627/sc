// Урок 10 · Повторення і практика: функції, List і Map — фінальна версія.
// Відкрий dartpad.dev, встав увесь код і натисни Run.
// Практика 1 — функції над списком · Практика 2 — Map-лічильник · Практика 3 — прайс і чек.

// ===== Практика 1: функції над списком температур =====

// Сума всіх чисел списку.
int sum(List<int> xs) {
  int s = 0;
  for (int x in xs) {
    s += x;
  }
  return s;
}

// Середнє: знак / завжди дає double.
double average(List<int> xs) => sum(xs) / xs.length;

// Найбільше: беремо перше і порівнюємо з рештою.
int maxOf(List<int> xs) {
  int m = xs[0];
  for (int x in xs) {
    if (x > m) m = x;
  }
  return m;
}

// Найменше — так само, тільки знак <.
int minOf(List<int> xs) {
  int m = xs[0];
  for (int x in xs) {
    if (x < m) m = x;
  }
  return m;
}

// Скільки днів тепліше за limit. Іменований параметр зі значенням за замовчуванням.
int countAbove(List<int> xs, {int limit = 15}) {
  int c = 0;
  for (int x in xs) {
    if (x > limit) c++;
  }
  return c;
}

// ⭐⭐ Самі значення, більші за limit, — фільтр where.
List<int> above(List<int> xs, int limit) => xs.where((x) => x > limit).toList();

// ===== Практика 2: Map-лічильник замовлень =====

// Скільки разів зустрілась кожна назва. Новий ключ — 1, наявний — +1.
Map<String, int> countAll(List<String> xs) {
  Map<String, int> counts = {};
  for (String x in xs) {
    counts.update(x, (v) => v + 1, ifAbsent: () => 1);
  }
  return counts;
}

// ===== Практика 3: прайс і чек =====

// Сума чека. Чого немає в прайсі — повідомляємо і пропускаємо.
// ⭐⭐ discount — знижка у відсотках, іменований параметр.
int total(List<String> cart, Map<String, int> prices, {int discount = 0}) {
  int s = 0;
  for (String item in cart) {
    if (prices.containsKey(item)) {
      s += prices[item] as int; // as int — як на уроці 6: «тут точно число»
    } else {
      print('Unknown: $item');
    }
  }
  return s * (100 - discount) ~/ 100;
}

void main() {
  // --- Практика 1 ---
  List<int> temps = [14, 17, 12, 19, 21, 16, 20]; // з понеділка по неділю
  print('Sum: ${sum(temps)}');
  print('Average: ${average(temps)}');
  print('Max: ${maxOf(temps)}, min: ${minOf(temps)}');
  print('Above 15: ${countAbove(temps)}');
  print('Above 18: ${countAbove(temps, limit: 18)}');
  print('Values above 18: ${above(temps, 18)}');
  // ⭐⭐⭐ Назва найтеплішого дня: номер максимуму в temps = номер дня в days.
  List<String> days = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
  print('Warmest: ${days[temps.indexOf(maxOf(temps))]}');

  // --- Практика 2 ---
  List<String> orders = ['coffee', 'tea', 'coffee', 'juice', 'coffee', 'tea', 'cocoa'];
  Map<String, int> counts = countAll(orders);
  counts.forEach((name, n) => print('$name: $n'));
  print('Different drinks: ${counts.length}');
  // Скільки всього замовлень — сума значень.
  int all = 0;
  for (int n in counts.values) {
    all += n;
  }
  print('All orders: $all');
  // Найпопулярніший — як чемпіон у турнірі.
  String top = '';
  int best = -1;
  counts.forEach((name, n) {
    if (n > best) {
      best = n;
      top = name;
    }
  });
  print('Top: $top');
  // ⭐⭐ Назви за абеткою: ключі -> список -> sort.
  List<String> names = counts.keys.toList();
  names.sort();
  print(names.join(', '));
  // ⭐⭐⭐ Те, що замовили лише раз.
  print(counts.keys.where((k) => counts[k] == 1).join(', '));

  // --- Практика 3 ---
  // Прайс — справжній словник: назва -> ціна.
  Map<String, int> prices = {'coffee': 45, 'tea': 30, 'juice': 40};
  prices['croissant'] = 55; // дописати
  List<String> cart = ['coffee', 'croissant', 'tea'];
  print('Total: ${total(cart, prices)}');
  cart.add('cocoa'); // цього в прайсі ще немає
  print('Total: ${total(cart, prices)}');
  prices['tea'] = 35; // змінити ціну
  prices['cocoa'] = 50; // додати позицію
  print('Total: ${total(cart, prices)}');
  prices.remove('juice'); // прибрати позицію
  print('Price list: ${prices.length} items');
  print('With 10% off: ${total(cart, prices, discount: 10)}');
  // ⭐⭐⭐ Виторг за день: кількість із Практики 2 × ціна з прайсу.
  int revenue = 0;
  counts.forEach((name, n) {
    if (prices.containsKey(name)) {
      revenue += n * (prices[name] as int);
    } else {
      print('No price: $name');
    }
  });
  print('Revenue: $revenue');
}

// Урок 10 · Підсумкова практика «Журнал оцінок» — фінальна версія ⭐–⭐⭐⭐.
// Відкрий dartpad.dev, встав увесь код і натисни Run.
// Журнал — словник: імʼя -> оцінка за контрольну (1–12).

// ⭐ Середній бал: сума значень / кількість.
double average(Map<String, int> marks) {
  int s = 0;
  for (int m in marks.values) {
    s += m;
  }
  return s / marks.length;
}

// ⭐ Найкращий: як чемпіон у турнірі — forEach і запамʼятати найбільше.
String best(Map<String, int> marks) {
  String name = '';
  int top = 0;
  marks.forEach((k, v) {
    if (v > top) {
      top = v;
      name = k;
    }
  });
  return name;
}

// ⭐⭐ Імена тих, у кого оцінка не нижча за limit.
List<String> atLeast(Map<String, int> marks, int limit) {
  List<String> names = [];
  marks.forEach((k, v) {
    if (v >= limit) names.add(k);
  });
  return names;
}

// ⭐⭐ Поставити оцінку з перевіркою: лише від 1 до 12.
void setMark(Map<String, int> marks, String name, int mark) {
  if (mark < 1 || mark > 12) {
    print('Invalid mark for $name: $mark');
    return;
  }
  marks[name] = mark;
}

// ⭐⭐⭐ Рівень за оцінкою: ланцюжок if / else if / else.
String level(int mark) {
  if (mark >= 10) {
    return 'high';
  } else if (mark >= 7) {
    return 'mid';
  } else {
    return 'low';
  }
}

// ⭐⭐⭐ Скільки учнів на кожному рівні — ще один словник-лічильник.
Map<String, int> levels(Map<String, int> marks) {
  Map<String, int> res = {};
  for (int m in marks.values) {
    res.update(level(m), (v) => v + 1, ifAbsent: () => 1);
  }
  return res;
}

void main() {
  Map<String, int> marks = {'Olha': 10, 'Ivan': 7, 'Nina': 12, 'Petro': 9, 'Sofiia': 11};

  // ⭐ Увесь журнал, середній і найкращий.
  marks.forEach((name, m) => print('$name: $m'));
  print('Average: ${average(marks)}');
  print('Best: ${best(marks)}');

  // ⭐⭐ Хто має 10 і більше · нові оцінки з перевіркою.
  print('10+: ${atLeast(marks, 10).join(', ')}');
  setMark(marks, 'Ivan', 8); // змінити наявну
  setMark(marks, 'Marko', 4); // дописати нового
  setMark(marks, 'Nina', 13); // не пройде перевірку
  print('Students: ${marks.length}, average: ${average(marks)}');

  // ⭐⭐⭐ Рівні.
  levels(marks).forEach((k, v) => print('$k: $v'));
}
