// Урок 17 · lib/main.dart першого проєкту — після практики 1.
// Це шаблон flutter create (лічильник) з нашими коментарями і трьома змінами,
// позначеними НОВЕ. Запуск: flutter run -d chrome або F5 у VS Code.
import 'package:flutter/material.dart';

// Точка входу. main() виконується один раз — hot reload його не перезапускає.
void main() {
  runApp(const MyApp());
}

// MyApp — рамка всього застосунку. StatelessWidget — сам не змінюється.
class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    // MaterialApp — корінь: назва вкладки, тема, перший екран (home).
    return MaterialApp(
      title: 'First app', // НОВЕ: назва вкладки в Chrome
      debugShowCheckedModeBanner: false, // прибрати стрічку DEBUG
      theme: ThemeData(
        colorScheme: .fromSeed(seedColor: Colors.teal), // НОВЕ: свій колір
      ),
      home: const MyHomePage(title: 'My first Flutter project'), // НОВЕ
    );
  }
}

// MyHomePage — екран зі станом. Сам стан — у класі _MyHomePageState.
class MyHomePage extends StatefulWidget {
  const MyHomePage({super.key, required this.title});

  final String title; // приходить від MyApp через конструктор

  @override
  State<MyHomePage> createState() => _MyHomePageState();
}

class _MyHomePageState extends State<MyHomePage> {
  int _counter = 0; // стан: після hot reload лишається, після hot restart — 0

  void _incrementCounter() {
    setState(() {
      _counter++;
    });
  }

  @override
  Widget build(BuildContext context) {
    // Scaffold — каркас сторінки: шапка (appBar), тіло (body), кругла кнопка.
    return Scaffold(
      // AppBar — шапка: колір, заголовок.
      appBar: AppBar(
        backgroundColor: Theme.of(context).colorScheme.inversePrimary,
        centerTitle: true, // НОВЕ: заголовок по центру
        title: Text(widget.title), // widget.title — поле з MyHomePage
      ),
      body: Center(
        child: Column(
          mainAxisAlignment: .center,
          children: [
            const Text('Clicks:'), // НОВЕ: замість довгого англійського речення
            Text(
              '$_counter',
              style: Theme.of(context).textTheme.headlineMedium,
            ),
          ],
        ),
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: _incrementCounter,
        tooltip: 'Increment',
        child: const Icon(Icons.add),
      ),
    );
  }
}
