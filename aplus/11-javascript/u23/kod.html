script.js · фінальний код уроку 
// Урок 23 · Міні-проєкт III «Вікторина на сторінці» · Прізвище Імʼя · рівень: ⭐

const questions = [
  { question: 'Яка гра — про будування з блоків?',
    options: ['Celeste', 'Minecraft', 'Portal 2'], answer: 'Minecraft' },
  { question: 'Скільки буде 7 * 8?',
    options: ['54', '56', '64'], answer: '56' },
  { question: 'Який тег робить посилання?',
    options: ['<a>', '<p>', '<img>'], answer: '<a>' },
  { question: 'Що повертає input.value?',
    options: ['число', 'рядок', 'масив'], answer: 'рядок' },
  { question: 'Яка подія спрацьовує на кожну літеру в полі?',
    options: ['click', 'change', 'input'], answer: 'input' },
];

const questionEl = document.querySelector('#question');
const optionsEl = document.querySelector('#options');
const msg = document.querySelector('#msg');
const scoreEl = document.querySelector('#score');

let score = 0;                       // поза функцією — памʼятає між кліками

// ── 1. Показати питання номер i: текст і по кнопці на кожен варіант ──
function showQuestion(i) {
  const q = questions[i];
  questionEl.textContent = q.question;  // textContent: '<a>' лишається текстом
  optionsEl.innerHTML = '';             // прибрати кнопки попереднього питання
  msg.textContent = '';

  for (const opt of q.options) {
    const btn = document.createElement('button');
    btn.textContent = opt;

    // ── 2. Обробник — ТУТ, усередині циклу: кожна кнопка знає свій opt ──
    btn.addEventListener('click', () => {
      if (opt === q.answer) {
        score++;
        msg.textContent = 'Правильно!';
      } else {
        msg.textContent = `Ні. Правильно: ${q.answer}`;
      }
      scoreEl.textContent = `Бали: ${score}`;
    });

    optionsEl.append(btn);
  }
}

showQuestion(0);
