List<int> xs = [4, 9, 2];

xs.insert(1, 5);
xs.remove(9);
print(xs);
print(xs.length);
print(xs.indexOf(2));
xs.where((x) => x > 3).length

2 · Що виведе: Map

Map<String, int> seats =
    {'A': 10, 'B': 4};
seats['C'] = 6;
seats['A'] = 8;
seats.remove('B');
print(seats);
print(seats['B']);
seats.containsKey('C')
