4 · style.css — нові правила уроку 
/* Картка — якір: усі absolute всередині рахують від її країв */
.card {
  position: relative;
}

/* Бейдж виходить з потоку і лягає поверх картинки */
.badge {
  position: absolute;
  top: 8px;              /* 8px від верхнього краю картки */
  right: 8px;            /* 8px від правого краю */
  z-index: 1;            /* просто число, без px */
  background: #af3b41;
  color: white;
  padding: 0.2rem 0.5rem;
  border-radius: 6px;
}

/* Другий бейдж: той самий клас badge + свій клас new */
.new {
  right: auto;           /* скасувати right від .badge */
  left: 8px;             /* лівий верхній кут */
  z-index: 2;            /* якщо зустрінуться — NEW зверху */
  background: green;
}

/* ⭐⭐⭐ Липка шапка: їде зі сторінкою, потім прилипає до верху */
header {
  position: sticky;
  top: 0;                /* без top не прилипає */
  z-index: 10;           /* вище за бейджі карток */
  background: #e2e7f6;   /* без фону текст сторінки просвічує */
}
