Функція бере масив — повертає число

функція з масивом

function total(arr) {
  let t = 0;
  for (const x of arr) t += x;
  return t;
}

return ≠ console.log

return      — для програми
console.log — для людини
без return  → undefined

функція кличе функцію

function half(arr) {
  return total(arr) / 2;
}
function first(arr) { return arr[0]; } console.log(first([8, 3]) + 1);

5 і undefined — два рядки: друкує всередині, а повертає нічого

Масив, який заповнює людина — тепер функцією

script.js · без перевірки оцінки
function readMarks() { const marks = []; let raw = prompt('Оцінка (порожньо — кінець):'); while (raw !== null && raw !== '') { marks.push(Number(raw)); raw = prompt('Ще оцінка:'); } return marks; }
Ввели 10, abc, 8 і порожньо[10, NaN, 8]
Скільки оцінок буде — невідомо, тому while
NaN і 13 у масив пускати не можна — потрібен if перед push
Перевірка: m >= 1 і m <= 12. Для NaN обидва порівняння дають false
Масив порожній — marks.length === 0. Середню не рахуємо: 0 / 0 → NaN

// marks = [10, 7, 12, 4, 9, 11, 8] function sum(arr) { let s = 0; for (const m of arr) { s += m; return s; } } function countAtLeast(arr, limit) { let c = 0; for (const m of arr) if (m > limit) c++; return c; } console.log('Сума:', sum); console.log('10 і вище:', countAtLeast(marks, 10));
КонсольСума: ƒ sum(arr) {…} 10 і вище: 2

Три біди — і жодної червоної

Знайди всі три, потім клацни

1. sum без дужок — друкує саму функцію. Треба sum(marks)

2. return s усередині циклу — функція виходить на першій оцінці: 10. return — після } циклу

3. m > limit — оцінка 10 не рахується. «10 і вище» — >=
