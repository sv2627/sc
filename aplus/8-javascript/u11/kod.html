// Урок 11 · Практична № 2 «Статистика класу» · рівень ⭐
// Прізвище Імʼя · рівень ⭐
const marks = [10, 7, 12, 9, 4, 11];   // заміни на 8 своїх оцінок

// сума: накопичувач до циклу, += у циклі, return після циклу
function sum(arr) {
  let s = 0;
  for (let i = 0; i < arr.length; i++) {
    s += arr[i];
  }
  return s;
}

// середнє: функція викликає іншу функцію
function average(arr) {
  return sum(arr) / arr.length;
}

console.log(`Сума: ${sum(marks)}`);                          // Сума: 53
console.log(`Середній бал: ${average(marks).toFixed(1)}`);   // Середній бал: 8.8
// i <= arr.length → читає arr[6], тобто undefined → сума NaN
⭐⭐ · max, min, countHigh 
// ⭐⭐ — допиши під кодом рівня ⭐, у тому самому script.js

// найвища: «рекорд» — перша оцінка, у циклі оновлюємо, якщо знайшли більшу
function max(arr) {
  let m = arr[0];
  for (let i = 1; i < arr.length; i++) {
    if (arr[i] > m) {
      m = arr[i];
    }
  }
  return m;
}

// найнижча: те саме, лише знак <
function min(arr) {
  let m = arr[0];
  for (let i = 1; i < arr.length; i++) {
    if (arr[i] < m) {
      m = arr[i];
    }
  }
  return m;
}

// скільки оцінок 10 і вище: лічильник
function countHigh(arr) {
  let n = 0;
  for (let i = 0; i < arr.length; i++) {
    if (arr[i] >= 10) {
      n++;
    }
  }
  return n;
}

console.log(`Найвища: ${max(marks)}`);         // Найвища: 12
console.log(`Найнижча: ${min(marks)}`);        // Найнижча: 4
console.log(`10 і вище: ${countHigh(marks)}`); // 10 і вище: 3
// let m = 0 у min → завжди 0: жодна оцінка не менша за нуль

// ⭐⭐⭐ — допиши під кодом рівня ⭐⭐

// новий масив лише з оцінками від 7: порожній res і push
function good(arr) {
  const res = [];
  for (let i = 0; i < arr.length; i++) {
    if (arr[i] >= 7) {
      res.push(arr[i]);
    }
  }
  return res;
}

// оцінки нижче 7 — так само, інша умова
function low(arr) {
  const res = [];
  for (let i = 0; i < arr.length; i++) {
    if (arr[i] < 7) {
      res.push(arr[i]);
    }
  }
  return res;
}

console.log(good(marks));   // [10, 7, 12, 9, 11]
console.log(low(marks));    // [4]
console.log(marks);         // [10, 7, 12, 9, 4, 11] — старий не змінився

// sort міняє сам масив; (a, b) => a - b — від меншої до більшої
marks.sort((a, b) => a - b);
console.log(marks);         // [4, 7, 9, 10, 11, 12]
marks.sort((a, b) => b - a);
console.log(marks);         // [12, 11, 10, 9, 7, 4]
// marks.sort() без функції порівнює як слова: [10, 11, 12, 4, 7, 9]
