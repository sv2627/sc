1 · Цикл for: відлік, таблиця множення, сума 
// Урок 8 · Практика 1 · цикл for
// for (старт; умова; крок) — коли знаємо, скільки разів повторити.

// Зворотний відлік: i починається з 5, поки i >= 1, щоразу мінус 1.
for (let i = 5; i >= 1; i--) {
  console.log(i);
}
console.log('Старт!'); // після } — виконається один раз

// Таблиця множення на n: i від 1 до 10, щоразу плюс 1 (i++).
const n = 7;
for (let i = 1; i <= 10; i++) {
  console.log(`${n} * ${i} = ${n * i}`);
}

// Сума від 1 до N. sum — ДО циклу, інакше щоразу ставала б нулем.
const N = Number(prompt('До якого числа?'));
let sum = 0;
for (let i = 1; i <= N; i++) {
  sum += i; // те саме, що sum = sum + i
}
console.log(`Сума від 1 до ${N}: ${sum}`); // 100 → 5050

// ⭐⭐ Парні числа від 2 до 20: крок не 1, а 2.
for (let i = 2; i <= 20; i += 2) {
  console.log(i);
}

// ⭐⭐⭐ Сходинки: рядок щоразу довшає на одну зірочку.
let line = '';
for (let i = 1; i <= 5; i++) {
  line += '*';
  console.log(line); // *  **  ***  ****  *****
}
2 · Цикл while: «Вгадай число» з лічильником і лімітом спроб 
// Урок 8 · Практика 2 · цикл while — гра «Вгадай число»
// while (умова) — повторюємо, ПОКИ умова правдива. Скільки разів — невідомо.

// Готовий рецепт: випадкове ціле число від 1 до 100.
const secret = Math.floor(Math.random() * 100) + 1;
// console.log(secret); // прибери // на початку — і підглянеш

let tries = 1; // ⭐⭐ перша спроба вже буде нижче
let guess = Number(prompt('Вгадай число від 1 до 100'));

// ⭐⭐⭐ не більше 7 спроб: обидві умови через &&
while (guess !== secret && tries < 7) {
  const hint = guess < secret ? 'Більше!' : 'Менше!';
  guess = Number(prompt(`${hint} Ще раз`));
  tries++; // ще одна спроба
}

// Після циклу: або вгадав, або спроби скінчились.
if (guess === secret) {
  alert(`Вгадав! Це ${secret}. Спроб: ${tries}`);
} else {
  alert(`Спроби скінчились. Це було ${secret}`);
}
3 · do-while, continue, break 
// Урок 8 · Практика 3 · do-while, break, continue

// do-while: спершу виконує тіло, потім перевіряє умову.
// Тому питання прозвучить хоч раз.
let answer;
do {
  answer = prompt('Скільки буде 7 * 8?');
} while (answer !== '56' && answer !== null); // null — натиснули Скасувати
alert(answer === null ? 'Шкода' : 'Правильно!');

// continue — пропустити цей крок і йти до наступного.
// Числа від 1 до 20 без кратних 3.
for (let i = 1; i <= 20; i++) {
  if (i % 3 === 0) continue;
  console.log(i);
}

// break — вийти з циклу зовсім.
// Перше число, що ділиться і на 7, і на 5.
for (let i = 1; i <= 100; i++) {
  if (i % 7 === 0 && i % 5 === 0) {
    console.log(i); // 35
    break;
  }
}

// ⭐⭐⭐ «Вгадай число», де Скасувати виходить із гри.
const secret = Math.floor(Math.random() * 100) + 1;
let guess = 0;
while (guess !== secret) {
  const raw = prompt('Число від 1 до 100');
  if (raw === null) break; // Скасувати — вийти з циклу
  guess = Number(raw);
  if (guess < secret) alert('Більше!');
  else if (guess > secret) alert('Менше!');
}
alert(guess === secret ? 'Вгадав!' : `Гру перервано. Це було ${secret}`);
