<!-- Крок 1–4: форма запису в клуб — на місці коментаря «Крок 1» -->
<form>
  <p>
    <label for="name">Ім'я</label>          <!-- for = id поля -->
    <input type="text" id="name" placeholder="Олена">
  </p>
  <p>
    <label for="mail">Email</label>
    <input type="email" id="mail" placeholder="olena@gmail.com">   <!-- без @ не пустить -->
  </p>
  <p>
    <label for="phone">Телефон</label>
    <input type="tel" id="phone" placeholder="+380 67 123 45 67">
  </p>
  <p>
    <label for="day">Дата першого заняття</label>
    <input type="date" id="day" min="2026-12-01">   <!-- ⭐⭐⭐ не раніше 1 грудня -->
  </p>
  <p>
    <label for="age">Вік</label>                       <!-- ⭐⭐ -->
    <input type="number" id="age" min="10" max="17">   <!-- ⭐⭐⭐ від 10 до 17 -->
  </p>
  <p>
    <label>Нік <input type="text" placeholder="Shadow"></label>   <!-- ⭐⭐ спосіб 2 -->
  </p>
  <p>
    <button>Записатися</button>                <!-- submit: перезавантажує сторінку -->
    <button type="reset">Очистити</button>     <!-- очищає всі поля -->
    <button type="button">Ще подумаю</button>  <!-- нічого, поки немає JavaScript -->
  </p>
</form>
4 · style.css — нові правила уроку 
/* Розминка: кнопка, що оживає */
.btn {
  transition: all 0.3s;        /* плавно і туди, і назад */
}
.btn:hover {
  background-color: #591429;
  color: white;
  transform: translateY(-3px); /* ⭐⭐ вгору на 3px */
}
