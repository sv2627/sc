Практика 1 · продажі за тиждень ⭐–⭐⭐⭐ 
// Урок 13 · Практика 1 «Продажі за тиждень» — фінальна версія ⭐–⭐⭐⭐.
// Скільки чашок кави продали щодня, з понеділка: порядок важливий -> List.

// Сума всіх чисел списку.
int total(List<int> xs) {
  int sum = 0;
  for (int x in xs) {
    sum += x;
  }
  return sum;
}

// Середнє. Ділення / дає дробове число -> тип double, не int.
double average(List<int> xs) => total(xs) / xs.length;

// Найбільше число: беремо перше і порівнюємо з кожним.
int best(List<int> xs) {
  int b = xs[0];
  for (int x in xs) {
    if (x > b) b = x;
  }
  return b;
}

// Скільки днів більше за limit: where відбирає, length рахує.
int daysAbove(List<int> xs, int limit) => xs.where((x) => x > limit).length;

// ⭐⭐⭐ Різниця з попереднім днем — цикл за номером, з другого дня (i = 1).
List<int> growth(List<int> xs) {
  List<int> res = [];
  for (int i = 1; i < xs.length; i++) {
    res.add(xs[i] - xs[i - 1]);
  }
  return res;
}

void main() {
  List<int> sales = [12, 7, 15, 9, 20, 3, 11];
  List<String> days = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
  print(total(sales)); // 77
  print(average(sales)); // 11.0
  print(best(sales)); // 20
  print(daysAbove(sales, 10)); // 4
  // ⭐⭐ День рекорду: номер найбільшого числа -> назва дня з тим самим номером.
  print(days[sales.indexOf(best(sales))]); // Fri
  print(growth(sales)); // [-5, 8, -6, 11, -17, 8]
}
Практика 2 · телефонна книга ⭐–⭐⭐⭐ 
// Урок 13 · Практика 2 «Телефонна книга» — фінальна версія ⭐–⭐⭐⭐.
// Номер шукаємо за іменем -> словник Map: ключ — імʼя, значення — номер.

// Номер за іменем. as String — «тут точно рядок», як as int на уроці 6.
String find(Map<String, String> book, String name) {
  if (book.containsKey(name)) return book[name] as String;
  return 'Unknown: $name';
}

// ⭐⭐ Додати контакт, лише якщо такого імені ще немає.
void addContact(Map<String, String> book, String name, String phone) {
  if (book.containsKey(name)) {
    print('Already exists: $name');
    return;
  }
  book[name] = phone;
}

// ⭐⭐⭐ Пошук навпаки: за номером — імʼя. Перебираємо ключі.
String whoIs(Map<String, String> book, String phone) {
  for (String name in book.keys) {
    if (book[name] == phone) return name;
  }
  return 'No one';
}

void main() {
  Map<String, String> phones = {
    'Olha': '067-111-22-33',
    'Ivan': '050-444-55-66',
  };
  phones['Nina'] = '063-777-88-99'; // новий ключ — новий запис
  print(phones['Ivan']); // 050-444-55-66

  print(find(phones, 'Nina')); // 063-777-88-99
  print(find(phones, 'Petro')); // Unknown: Petro
  phones['Ivan'] = '050-000-00-00'; // ключ уже є — значення замінюється
  phones.remove('Olha');
  print(phones.length); // 2
  phones.forEach((name, phone) => print('$name: $phone'));

  addContact(phones, 'Nina', '099-000-00-00'); // Already exists: Nina
  addContact(phones, 'Anna', '097-123-45-67');
  // ⭐⭐ Імена за абеткою: ключі -> список -> sort.
  List<String> names = phones.keys.toList();
  names.sort();
  print(names); // [Anna, Ivan, Nina]

  print(whoIs(phones, '050-000-00-00')); // Ivan
  print(whoIs(phones, '000')); // No one
}
Кавʼярня: день у касі · фінальна версія ⭐–⭐⭐⭐ 
// Урок 13 · Маленький проєкт «Кавʼярня: день у касі» — фінальна версія ⭐–⭐⭐⭐.
// Замовлення — СУТНІСТЬ (у кожного той самий набір полів і дій) → class Order.
// Меню «назва → ціна» і підрахунки «назва → скільки» — СЛОВНИКИ → Map.

// ⭐ Клас замовлення: хто замовив і що саме (список позицій).
class Order {
  String customer;
  List<String> items;
  Order(this.customer, this.items);

  // Сума замовлення за меню. Позиції, якої немає в меню, не рахуємо.
  int total(Map<String, int> menu) {
    int sum = 0;
    for (String item in items) {
      if (menu.containsKey(item)) {
        sum += menu[item] as int; // as int — «тут точно число», як на уроці 6
      }
    }
    return sum;
  }

  // Один рядок для друку: Olha: coffee, croissant = 100 grn
  String info(Map<String, int> menu) =>
      '$customer: ${items.join(', ')} = ${total(menu)} grn';
}

// ⭐⭐ Виторг за день — сума всіх замовлень.
int revenue(List<Order> day, Map<String, int> menu) {
  int sum = 0;
  for (Order o in day) {
    sum += o.total(menu);
  }
  return sum;
}

// ⭐⭐ Найбільше замовлення — функція повертає цілий обʼєкт.
Order biggest(List<Order> day, Map<String, int> menu) {
  Order best = day[0];
  for (Order o in day) {
    if (o.total(menu) > best.total(menu)) best = o;
  }
  return best;
}

// ⭐⭐ Скільки продано кожної позиції: «назва → кількість».
Map<String, int> soldCounts(List<Order> day) {
  Map<String, int> sold = {};
  for (Order o in day) {
    for (String item in o.items) {
      sold.update(item, (v) => v + 1, ifAbsent: () => 1);
    }
  }
  return sold;
}

// ⭐⭐⭐ Хіт дня — позиція з меню, якої продано найбільше.
String bestSeller(Map<String, int> sold, Map<String, int> menu) {
  String best = '';
  int max = 0;
  sold.forEach((item, n) {
    if (menu.containsKey(item) && n > max) {
      best = item;
      max = n;
    }
  });
  return best;
}

// ⭐⭐⭐ Скільки витратив кожен гість — «імʼя → гривні».
Map<String, int> spentBy(List<Order> day, Map<String, int> menu) {
  Map<String, int> spent = {};
  for (Order o in day) {
    int t = o.total(menu);
    spent.update(o.customer, (v) => v + t, ifAbsent: () => t);
  }
  return spent;
}

void main() {
  // Меню — словник: назва → ціна, грн.
  Map<String, int> menu = {'coffee': 45, 'tea': 30, 'croissant': 55, 'juice': 40};

  // ⭐ Замовлення за день — список обʼєктів.
  List<Order> day = [
    Order('Olha', ['coffee', 'croissant']),
    Order('Ivan', ['tea']),
    Order('Nina', ['coffee', 'coffee', 'juice']),
    Order('Olha', ['tea', 'cocoa']),
    Order('Marko', ['juice', 'croissant', 'coffee']),
  ];

  // ⭐ Усі замовлення і скільки їх.
  for (Order o in day) {
    print(o.info(menu));
  }
  print('Orders: ${day.length}');

  // ⭐⭐ Виторг, найбільше замовлення, продано по позиціях.
  print('Revenue: ${revenue(day, menu)} grn');
  Order big = biggest(day, menu);
  print('Biggest: ${big.customer}, ${big.total(menu)} grn');
  Map<String, int> sold = soldCounts(day);
  sold.forEach((item, n) => print('$item: $n'));

  // ⭐⭐⭐ Чого немає в меню, хіт дня, середній чек, витрати гостей.
  for (String item in sold.keys) {
    if (!menu.containsKey(item)) print('Not on menu: $item');
  }
  print('Best seller: ${bestSeller(sold, menu)}');
  print('Average check: ${revenue(day, menu) / day.length} grn');
  spentBy(day, menu).forEach((name, s) => print('$name spent $s grn'));
}
