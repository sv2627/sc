1 · Створити, прочитати, змінити 
// Урок 9 · Практика 1 · масив
// Масив — список значень у [ ], між ними коми. Індекс (номер місця) — з 0.
const games = ['Minecraft', 'Terraria', 'Celeste', 'Portal', 'Tetris'];

console.log(games[0]);       // Minecraft — перша
console.log(games[2]);       // Celeste — третя (індекс 2!)
console.log(games.length);   // 5 — скільки елементів

games[1] = 'Stardew Valley'; // замінили другу
console.log(games);

console.log(games[10]);      // undefined — такого місця немає, але це не помилка

// ⭐⭐ Остання — індекс length - 1, працює з будь-якою довжиною
console.log(games[games.length - 1]);   // Tetris

// ⭐⭐⭐ Поміняти першу й останню місцями через допоміжну змінну
const last = games.length - 1;
const temp = games[0];
games[0] = games[last];
games[last] = temp;
console.log(games);   // ['Tetris', 'Stardew Valley', 'Celeste', 'Portal', 'Minecraft']
2 · Кошик росте: push, pop, includes, join 
// Урок 9 · Практика 2 · push і родина
const cart = [];              // порожній кошик
cart.push('Terraria');        // push — додати в кінець
cart.push('Celeste');
cart.push('Portal');
console.log(cart.join(', '));            // Terraria, Celeste, Portal
console.log(`Ігор: ${cart.length}`);     // Ігор: 3

const removed = cart.pop();   // pop — забрати останній і віддати його
console.log(`Забрали: ${removed}, лишилось: ${cart.length}`);   // Portal, 2

// ⭐⭐ Додати, тільки якщо такої гри ще немає
if (cart.includes('Minecraft')) {
  console.log('Minecraft уже в кошику');
} else {
  cart.push('Minecraft');
}
console.log(cart.join(', '));   // Terraria, Celeste, Minecraft

3 · Кошик GameStore: сума, середнє, лічильник, найдорожча 
// Урок 9 · Практика 3 · «Кошик GameStore»: масив цін → сума
const prices = [699, 199, 449, 319, 250];
prices.push(99);   // ще одна ціна ДО циклу — сума врахує її сама

let sum = 0;       // накопичувач — до циклу
for (let i = 0; i < prices.length; i++) {   // <, не <= : останній індекс length - 1
  sum += prices[i];
}
const avg = sum / prices.length;
console.log(`Разом: ${sum} грн`);        // Разом: 2015 грн
console.log(`Середня: ${avg.toFixed(2)} грн`);   // Середня: 335.83 грн

// Скільки ігор дешевших за 300 грн — лічильник
let cheap = 0;
for (let i = 0; i < prices.length; i++) {
  if (prices[i] < 300) {
    cheap++;
  }
}
console.log(`Дешевших за 300: ${cheap}`);   // 3 (199, 250, 99)

// ⭐⭐ Найдорожча: стартуємо з першої, далі «бачу більшу — запамʼятовую»
let max = prices[0];
for (let i = 0; i < prices.length; i++) {
  if (prices[i] > max) {
    max = prices[i];
  }
}
console.log(`Найдорожча: ${max} грн`);   // 699

// ⭐⭐⭐ Два масиви поруч: той самий індекс — та сама гра
const games = ['Minecraft', 'Terraria', 'Celeste'];
const gamePrices = [699, 199, 449];
for (let i = 0; i < games.length; i++) {
  console.log(`${games[i]} — ${gamePrices[i]} грн`);
}
